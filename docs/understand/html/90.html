19     class Nivel(StrEnum):
20         NORMAL = "NORMAL"
21         PRECAUCION = "PRECAUCION"
22         CRITICA = "CRITICA"
23     
24     
25     NOMBRES_PARAMETRO = {
26         Parametro.TEMPERATURA: "Temperatura",
27         Parametro.OXIGENO_DISUELTO: "Oxígeno disuelto",
28         Parametro.PH: "pH",
29     }
30     NOMBRES_NIVEL = {Nivel.NORMAL: "normal", Nivel.PRECAUCION: "PRECAUCIÓN", Nivel.CRITICA: "CRÍTICA"}

33     def _dentro(valor: float, minimo: float | None, maximo: float | None) -> bool:
34         return (minimo is None or valor >= minimo) and (maximo is None or valor <= maximo)
35     
36     
37     def _rango(minimo: float | None, maximo: float | None) -> str:
38         if minimo is None and maximo is None:
39             return "cualquier valor"
40         if minimo is None:
41             return f"<= {maximo:g}"
42         if maximo is None:
43             return f">= {minimo:g}"
44         return f"{minimo:g}–{maximo:g}"

47     @dataclass(frozen=True)
48     class Umbral:
49         parametro: Parametro
50         unidad: str
51         normal_min: float | None
52         normal_max: float | None
53         precaucion_min: float | None
54         precaucion_max: float | None
55     
56         def __post_init__(self) -> None:
57             def contiene(exterior: float | None, interior: float | None, es_min: bool) -> bool:
58                 if exterior is None:
59                     return True
60                 if interior is None:
61                     return False
62                 return exterior <= interior if es_min else exterior >= interior
63     
64             if not (
65                 contiene(self.precaucion_min, self.normal_min, True)
66                 and contiene(self.precaucion_max, self.normal_max, False)
67             ):
68                 raise ErrorValidacion(
69                     self.parametro.value.lower(),
70                     f"Umbral inválido para {self.parametro.value}: el rango de precaución "
71                     "debe contener al rango normal",
72                 )
73             for a, b in (
74                 (self.normal_min, self.normal_max),
75                 (self.precaucion_min, self.precaucion_max),
76             ):
77                 if a is not None and b is not None and a > b:
78                     raise ErrorValidacion(
79                         self.parametro.value.lower(),
80                         f"Umbral inválido para {self.parametro.value}: mínimo mayor que máximo",
81                     )
82     
83         def clasificar(self, valor: float) -> Nivel:
84             if _dentro(valor, self.normal_min, self.normal_max):
85                 return Nivel.NORMAL
86             if _dentro(valor, self.precaucion_min, self.precaucion_max):
87                 return Nivel.PRECAUCION
88             return Nivel.CRITICA
89     
90         @property
91         def rango_normal(self) -> str:
92             return _rango(self.normal_min, self.normal_max)
93     
94         def describir(self, valor: float, nivel: Nivel) -> str:
95             unidad = f" {self.unidad}" if self.unidad and self.unidad != "pH" else ""
96             return (
97                 f"{NOMBRES_PARAMETRO[self.parametro]} en {NOMBRES_NIVEL[nivel]}: "
98                 f"{valor:.2f}{unidad} (normal {self.rango_normal}{unidad})"
99             )

102     @dataclass(frozen=True)
103     class ConjuntoUmbrales:
104         temperatura: Umbral
105         oxigeno_disuelto: Umbral
106         ph: Umbral
107     
108         def __post_init__(self) -> None:
109             for p, u in self.por_parametro().items():
110                 if u.parametro is not p:
111                     raise ErrorValidacion(p.value.lower(), f"El umbral de {p.value} está mal asignado")
112     
113         def por_parametro(self) -> dict[Parametro, Umbral]:
114             return {
115                 Parametro.TEMPERATURA: self.temperatura,
116                 Parametro.OXIGENO_DISUELTO: self.oxigeno_disuelto,
117                 Parametro.PH: self.ph,
118             }
119     
120         def de(self, parametro: Parametro) -> Umbral:
121             return self.por_parametro()[parametro]
