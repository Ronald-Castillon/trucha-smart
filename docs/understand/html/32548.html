1     """Ejecuta las pruebas unitarias y de integración y deja coverage-sonar.xml para SonarQube.
2     
3     coverage.py escribe en <source> la ruta absoluta de la máquina (p. ej. D:/.../src); el escáner
4     corre en un contenedor con el proyecto en /usr/src, así que la ruta se reescribe a «src»,
5     relativa a la raíz del proyecto, que SonarQube resuelve en cualquier máquina.
6     """
7     
8     from __future__ import annotations
9     
10     import re
11     import subprocess
12     import sys
13     from pathlib import Path
14     
15     RAIZ = Path(__file__).resolve().parents[1]
16     SALIDA = RAIZ / "coverage-sonar.xml"
17     
18     
19     def main() -> int:
20         r = subprocess.run(
21             [sys.executable, "-m", "pytest", "tests/unit", "tests/integration", "-q",
22              "--cov=src", f"--cov-report=xml:{SALIDA}"],
23             cwd=RAIZ,
24         )  # fmt: skip
25         if r.returncode != 0:
26             return r.returncode
27         xml = SALIDA.read_text(encoding="utf-8")
28         xml = re.sub(r"<source>[^<]*</source>", "<source>src</source>", xml)
29         SALIDA.write_text(xml, encoding="utf-8")
30         print(f"Cobertura para SonarQube en {SALIDA.name}")
31         return 0
32     
33     
34     if __name__ == "__main__":
35         raise SystemExit(main())
