23     class EvaluadorCalidadAgua:
24         def __init__(self, umbrales: ConjuntoUmbrales) -> None:
25             self._umbrales = umbrales
26     
27         def evaluar(self, lectura: LecturaAgua) -> list[Evaluacion]:
28             """Una evaluación por parámetro, en el orden temperatura, oxígeno, pH."""
29             resultado = []
30             for parametro, valor in lectura.valores().items():
31                 umbral = self._umbrales.de(parametro)
32                 nivel = umbral.clasificar(valor)
33                 resultado.append(Evaluacion(parametro, valor, nivel, umbral.describir(valor, nivel)))
34             return resultado
35     
36         def niveles(self, lectura: LecturaAgua) -> dict[Parametro, Nivel]:
37             return {e.parametro: e.nivel for e in self.evaluar(lectura)}
