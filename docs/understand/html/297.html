89     @admin.post("", response_model=ClaveCreadaSalida, status_code=201)
90     def crear_clave(
91         datos: ClaveEntrada,
92         request: Request,
93         usuario: Usuario = Depends(usuario_actual),
94         c: Contenedor = Depends(contenedor),
95     ) -> ClaveCreadaSalida:
96         """Emite una clave de integración (solo Administrador). La clave se devuelve una única vez."""
97         creada = c.integraciones.crear(datos.nombre, usuario, _ip(request))
98         c.confirmar()
99         return ClaveCreadaSalida(**ClaveSalida.de(creada.clave).model_dump(), clave=creada.secreto)

102     @admin.delete("/{clave_id}", response_model=ClaveSalida)
103     def revocar_clave(
104         clave_id: int,
105         request: Request,
106         usuario: Usuario = Depends(usuario_actual),
107         c: Contenedor = Depends(contenedor),
108     ) -> ClaveSalida:
109         revocada = c.integraciones.revocar(clave_id, usuario, _ip(request))
110         c.confirmar()
111         return ClaveSalida.de(revocada)

117     def clave_actual(
118         request: Request,
119         secreto: str | None = Security(_cabecera_clave),
120         c: Contenedor = Depends(contenedor),
121     ) -> ClaveIntegracion:
122         ip = _ip(request)
123         try:
124             clave = c.integraciones.autenticar(secreto, ip)
125         finally:
126             c.confirmar()  # registra el último uso o el rechazo en la bitácora (RF-11)
127         return clave

131     def estanques_externos(
132         clave: ClaveIntegracion = Depends(clave_actual), c: Contenedor = Depends(contenedor)
133     ) -> list[EstanqueExterno]:
134         """Estado actual de los estanques activos: última lectura, nivel por parámetro y alertas."""
135         return [
136             EstanqueExterno(
137                 id=e.estanque.id,
138                 nombre=e.estanque.nombre,
139                 tipo=e.estanque.tipo.value,
140                 ultima_lectura=_lectura(e.ultima_lectura) if e.ultima_lectura else None,
141                 niveles=e.niveles,
