1     """WCAG 4.1.2 y 1.4.1: árbol de accesibilidad que recibe un lector de pantalla (Chromium) en la
2     app desplegada, y textos que acompañan a cada color de estado.
3     
4     Uso: docker compose up -d --build && python tools/evidencias/aria.py login|panel|color
5     """
6     
7     import re
8     import sys
9     
10     from playwright.sync_api import sync_playwright
11     
12     sys.stdout.reconfigure(encoding="utf-8")
13     URL = "http://localhost:8000"
14     modo = sys.argv[1]
15     with sync_playwright() as p:
16         nav = p.chromium.launch()
17         pag = nav.new_page(viewport={"width": 1366, "height": 900})
18         pag.goto(f"{URL}/login")
19         if modo == "login":
20             print("Árbol de accesibilidad de /login (lo que anuncia un lector de pantalla):\n")
21             print(pag.locator("body").aria_snapshot())
22         pag.fill("input[name=nombre_usuario]", "operario")
23         pag.fill("input[name=password]", "Oper123!")
24         pag.click("button[type=submit]")
25         pag.wait_for_url("**/panel")
26         pag.wait_for_timeout(1200)
27         if modo == "panel":
28             print(
29                 "Árbol de accesibilidad del panel: enlace de salto, navegación y tarjeta de la Jaula J1:\n"
30             )
31             arbol = pag.locator("body").aria_snapshot()
32             lineas = arbol.splitlines()
33             fin_nav = next(i for i, ln in enumerate(lineas) if "main" in ln)
34             print("\n".join(lineas[: fin_nav + 1]))
35             print("  …")
36             print(pag.locator('[data-estanque="3"]').aria_snapshot())
37         if modo == "color":
38             print(
39                 "WCAG 1.4.1: cada color de estado va acompañado de texto (no se depende solo del color)"
40             )
41             for ruta in ("/panel", "/alertas"):
42                 pag.goto(f"{URL}{ruta}")
43                 pag.wait_for_timeout(800)
44                 print(f"\n{ruta}")
45                 vistos = set()
46                 for el in pag.locator(".etiqueta").all():
47                     clase = (el.get_attribute("class") or "").replace("etiqueta", "").strip()
48                     color = el.evaluate("e => getComputedStyle(e).backgroundColor")
49                     texto = re.sub(r"\s+", " ", el.inner_text()).strip()[:40]
50                     if (clase, texto) in vistos or not texto:
51                         continue
52                     vistos.add((clase, texto))
53                     print(f"  clase={clase:<22} fondo={color:<20} texto visible=«{texto}»")
54         nav.close()
