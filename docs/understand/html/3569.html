1     """Recomendación de ración diaria (RF-07, brecha B-08).
2     
3       ración (kg/día) = biomasa (kg) x tasa de alimentación (% biomasa/día según peso medio)
4                         x factor de temperatura x factor de oxígeno
5     La tabla y los factores vienen de config/alimentacion.yaml (ADR-005). Un factor 0 (nivel
6     CRÍTICO) significa suspender la alimentación. La recomendación siempre muestra la regla
7     aplicada; la decisión final es del piscicultor.
8     """
9     
10     from __future__ import annotations
11     
12     from dataclasses import dataclass, field
13     
14     from src.domain.errores import ErrorValidacion
15     from src.domain.model.umbral import NOMBRES_NIVEL, Nivel
16     
17     
18     @dataclass(frozen=True)
19     class TramoTasa:
20         desde_g: float
21         hasta_g: float | None  # None = sin límite superior
22         tasa_pct: float
23     
24         def contiene(self, peso_g: float) -> bool:
25             return peso_g >= self.desde_g and (self.hasta_g is None or peso_g < self.hasta_g)
26     
27         @property
28         def descripcion(self) -> str:
29             if self.hasta_g is None:
30                 return f">= {self.desde_g:g} g"
31             if self.desde_g == 0:
32                 return f"< {self.hasta_g:g} g"
33             return f"{self.desde_g:g}–{self.hasta_g:g} g"
34     
35     
36     @dataclass(frozen=True)
37     class TablaAlimentacion:
38         tramos: tuple[TramoTasa, ...]
39         factor_temperatura: dict[Nivel, float] = field(hash=False)
40         factor_oxigeno: dict[Nivel, float] = field(hash=False)
41     
42         def __post_init__(self) -> None:
43             if not self.tramos or self.tramos[0].desde_g != 0:
44                 raise ErrorValidacion("tasas", "La tabla de alimentación debe empezar en 0 g")
45             for previo, siguiente in zip(self.tramos, self.tramos[1:], strict=False):
46                 if previo.hasta_g != siguiente.desde_g:
47                     raise ErrorValidacion("tasas", "Los tramos de peso deben ser contiguos")
48             if self.tramos[-1].hasta_g is not None:
49                 raise ErrorValidacion("tasas", "El último tramo debe no tener límite superior")
50             for t in self.tramos:
51                 if t.tasa_pct <= 0:
52                     raise ErrorValidacion("tasas", "Las tasas de alimentación deben ser > 0 %")
53             for nombre, factores in (
54                 ("factor_temperatura", self.factor_temperatura),
55                 ("factor_oxigeno", self.factor_oxigeno),
56             ):
57                 if set(factores) != set(Nivel):
58                     raise ErrorValidacion(nombre, f"{nombre} debe definir normal, precaucion y critica")
59                 if any(not 0 <= f <= 1 for f in factores.values()):
60                     raise ErrorValidacion(nombre, f"{nombre} debe estar entre 0 y 1")
61     
62         def tramo_para(self, peso_g: float) -> TramoTasa:
63             return next(t for t in self.tramos if t.contiene(peso_g))
64     
65     
66     @dataclass(frozen=True)
67     class Recomendacion:
68         biomasa_kg: float
69         peso_medio_g: float
70         tasa_pct: float
71         tramo: str
72         nivel_temperatura: Nivel | None
73         nivel_oxigeno: Nivel | None
74         factor_temperatura: float
75         factor_oxigeno: float
76         racion_kg: float
77         suspender: bool
78         regla: str
79         advertencias: tuple[str, ...] = ()
80     
81     
82     class RecomendadorRacion:
83         def __init__(self, tabla: TablaAlimentacion) -> None:
84             self._tabla = tabla
85     
86         def recomendar(
87             self,
88             biomasa_kg: float,
89             peso_medio_g: float,
90             nivel_temperatura: Nivel | None,
91             nivel_oxigeno: Nivel | None,
92         ) -> Recomendacion:
93             if biomasa_kg < 0:
94                 raise ErrorValidacion("biomasa_kg", "La biomasa no puede ser negativa")
95             if peso_medio_g <= 0:
96                 raise ErrorValidacion("peso_medio_g", "El peso medio debe ser mayor que 0")
97             advertencias = []
98             if nivel_temperatura is None or nivel_oxigeno is None:
99                 advertencias.append(
100                     "Sin lectura de calidad de agua: se asumen factores normales; "
101                     "mida temperatura y oxígeno antes de alimentar"
102                 )
103             n_t = nivel_temperatura or Nivel.NORMAL
104             n_o = nivel_oxigeno or Nivel.NORMAL
105             tramo = self._tabla.tramo_para(peso_medio_g)
106             f_t = self._tabla.factor_temperatura[n_t]
107             f_o = self._tabla.factor_oxigeno[n_o]
108             racion = round(biomasa_kg * tramo.tasa_pct / 100 * f_t * f_o, 3)
109             suspender = f_t == 0 or f_o == 0
110             regla = (
111                 f"Ración = {biomasa_kg:.3f} kg de biomasa × {tramo.tasa_pct:g} % "
112                 f"(peso medio {peso_medio_g:g} g, tramo {tramo.descripcion}) "
113                 f"× {f_t:g} (temperatura {NOMBRES_NIVEL[n_t]}) "
114                 f"× {f_o:g} (oxígeno {NOMBRES_NIVEL[n_o]}) = {racion:.3f} kg/día"
115             )
116             if suspender:
117                 motivos = [
118                     nombre
119                     for nombre, nivel in (("temperatura", n_t), ("oxígeno disuelto", n_o))
120                     if nivel is Nivel.CRITICA
121                 ]
122                 regla = f"Suspender alimentación: {' y '.join(motivos)} en nivel CRÍTICO. " + regla
123             return Recomendacion(
124                 biomasa_kg=biomasa_kg,
125                 peso_medio_g=peso_medio_g,
126                 tasa_pct=tramo.tasa_pct,
127                 tramo=tramo.descripcion,
128                 nivel_temperatura=nivel_temperatura,
129                 nivel_oxigeno=nivel_oxigeno,
130                 factor_temperatura=f_t,
131                 factor_oxigeno=f_o,
132                 racion_kg=racion,
133                 suspender=suspender,
134                 regla=regla,
135                 advertencias=tuple(advertencias),
136             )
