1     """Mide los indicadores de valor TO-BE (IV-1, IV-2, IV-3) sobre la aplicación en ejecución.
2     
3     Levanta la app (Uvicorn) de un checkout del repositorio sobre una SQLite temporal con

9             la API. Si la versión no tiene reportes (Inc. 1) queda en null.
10       IV-3  Lecturas registradas por estanque y día: se ejecuta el simulador durante una ventana
11             corta con el intervalo indicado y se cuenta lo que quedó en la BD; el resultado se
12             extrapola a 24 h (se guarda la ventana real para que quede explícito).
13     
14     Escribe docs/gqm/mediciones/medicion_<etiqueta>.json con las muestras crudas.
15     
16     Uso:  python tools/medir_indicadores.py --etiqueta inc2

46     def _token(api: httpx.Client, usuario: str) -> dict[str, str]:
47         r = api.post("/api/auth/login", json={"nombre_usuario": usuario, "password": USUARIOS[usuario]})
48         r.raise_for_status()
49         return {"Authorization": f"Bearer {r.json()['access_token']}"}

52     def medir_iv1(url: str, repeticiones: int) -> dict:
53         from playwright.sync_api import sync_playwright
54     
55         with httpx.Client(base_url=url, timeout=10) as api:
56             pisc, oper = _token(api, "piscicultor"), _token(api, "operario")
57             ids = [
58                 api.post("/api/estanques", json={"nombre": f"Medición IV-1 #{i}"}, headers=pisc).json()[
59                     "id"
60                 ]
61                 for i in range(repeticiones)
62             ]
63             muestras = []
64             with sync_playwright() as p:
65                 nav = p.chromium.launch()
66                 pag = nav.new_page(base_url=url)
67                 pag.goto("/login")
68                 pag.fill("input[name=nombre_usuario]", "operario")
69                 pag.fill("input[name=password]", USUARIOS["operario"])
70                 pag.click("button[type=submit]")
71                 pag.wait_for_url("**/panel")
72                 for eid in ids:
73                     inicio = time.perf_counter()
74                     api.post(
75                         "/api/lecturas",
76                         json={"estanque_id": eid, "temperatura_c": 12.5, "oxigeno_mg_l": 4.0,
77                               "ph": 7.3},
78                         headers=oper,
79                     ).raise_for_status()  # fmt: skip
80                     pag.locator(f'[data-estanque="{eid}"] .alerta .etiqueta.CRITICA').wait_for(
81                         timeout=30000
82                     )
83                     muestras.append(round(time.perf_counter() - inicio, 3))
84                 nav.close()
85         return {
86             "descripcion": "lectura crítica (OD 4.0 mg/L) por API -> alerta CRÍTICA visible en el "
87             "panel web abierto",
88             "muestras_s": muestras,
89             "promedio_s": round(sum(muestras) / len(muestras), 3),
90             "maximo_s": max(muestras),
91             "valor_minutos": round(max(muestras) / 60, 4),
92             "criterio": "máximo de las muestras (peor caso observado)",
93         }

96     def medir_iv2(url: str) -> dict:
97         with httpx.Client(base_url=url, timeout=60) as api:
98             pisc = _token(api, "piscicultor")
99             rutas = [
100                 f"/api/reportes/{tipo}.{fmt}?desde=2026-10-01&hasta=2026-10-31"
101                 for tipo in ("calidad-agua", "produccion")
102                 for fmt in ("pdf", "csv")
103             ]
104             inicio = time.perf_counter()
105             detalle = []
106             for ruta in rutas:
107                 r = api.get(ruta, headers=pisc)
108                 if r.status_code == 404:
109                     return {"descripcion": "la versión no tiene exportación de reportes",
110                             "valor_horas": None}  # fmt: skip
111                 r.raise_for_status()
112                 detalle.append({"ruta": ruta, "bytes": len(r.content)})
113             segundos = round(time.perf_counter() - inicio, 3)
114         return {
115             "descripcion": "obtener el reporte mensual (octubre 2026): calidad de agua y producción "
116             "en PDF y CSV por la API",
117             "archivos": detalle,
118             "segundos": segundos,
119             "valor_horas": round(segundos / 3600, 6),
120         }

123     def medir_iv3(url: str, repo: Path, env: dict, intervalo: float, cantidad: int) -> dict:
124         with httpx.Client(base_url=url, timeout=10) as api:
125             pisc = _token(api, "piscicultor")
126             eid = api.post("/api/estanques", json={"nombre": "Medición IV-3"}, headers=pisc).json()[
127                 "id"
128             ]
129             inicio = time.perf_counter()
130             subprocess.run(
131                 [sys.executable, "-m", "src.adapters.inbound.simulator", "--estanque", str(eid),
132                  "--escenario", "normal", "--intervalo", str(intervalo), "--cantidad", str(cantidad),
133                  "--url", url],
134                 cwd=repo, env=env, check=True, capture_output=True,
135             )  # fmt: skip
136             ventana = time.perf_counter() - inicio
137             oper = _token(api, "operario")
138             registradas = len(
139                 api.get(f"/api/lecturas?estanque_id={eid}&limite=500", headers=oper).json()
140             )
141         por_dia = registradas / ventana * 86400
142         return {
143             "descripcion": f"simulador escenario normal con --intervalo {intervalo:g} s; lecturas "
144             "registradas en la BD durante la ventana, extrapoladas a 24 h",
145             "lecturas_registradas": registradas,
146             "ventana_s": round(ventana, 3),
147             "valor_lecturas_dia": round(por_dia, 1),
148         }

151     def main(argv: list[str] | None = None) -> int:
152         parser = argparse.ArgumentParser(description="Medición TO-BE de IV-1..IV-3")
153         parser.add_argument("--etiqueta", required=True, help="p. ej. inc1, inc2")
154         parser.add_argument("--repo", type=Path, default=RAIZ, help="checkout a medir")
155         parser.add_argument("--repeticiones", type=int, default=5)
156         parser.add_argument("--intervalo", type=float, default=5.0)
157         parser.add_argument("--cantidad", type=int, default=7)
158         args = parser.parse_args(argv)
159         if hasattr(sys.stdout, "reconfigure"):
160             sys.stdout.reconfigure(encoding="utf-8")
161         repo = args.repo.resolve()
162     
163         with tempfile.TemporaryDirectory(ignore_cleanup_errors=True) as tmp:
164             bd = f"sqlite:///{Path(tmp, 'medicion.db').as_posix()}"
165             env = {**os.environ, "DATABASE_URL": bd, "JWT_SECRET": "medicion-indicadores-32-bytes-min!",
166                    "PYTHONIOENCODING": "utf-8"}  # fmt: skip
167             subprocess.run([sys.executable, "tools/init_db.py", "--url", bd], cwd=repo, env=env,
168                            check=True, capture_output=True)  # fmt: skip
169             puerto = _puerto()
170             url = f"http://127.0.0.1:{puerto}"
171             servidor = subprocess.Popen(
172                 [sys.executable, "-m", "uvicorn", "src.main:app", "--port", str(puerto)],
173                 cwd=repo, env=env, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
174             )  # fmt: skip
175             try:
176                 for _ in range(100):
177                     try:
178                         if httpx.get(f"{url}/api/salud", timeout=0.5).status_code == 200:
179                             break
180                     except httpx.HTTPError:
181                         time.sleep(0.1)
182                 version = subprocess.run(
183                     ["git", "describe", "--tags", "--always"], cwd=repo, capture_output=True,
184                     text=True,
185                 ).stdout.strip()  # fmt: skip
186                 resultado = {
187                     "etiqueta": args.etiqueta,
188                     "version_medida": version,
189                     "medido_en": datetime.now(UTC).isoformat(timespec="seconds"),
190                     "entorno": f"{sys.platform}, Python {sys.version.split()[0]}, SQLite, Chromium "
191                     "headless (Playwright)",
192                     "IV-1": medir_iv1(url, args.repeticiones),
193                     "IV-2": medir_iv2(url),
194                     "IV-3": medir_iv3(url, repo, env, args.intervalo, args.cantidad),
195                 }
196             finally:
197                 servidor.terminate()
198                 servidor.wait(timeout=10)
199     
200         SALIDA.mkdir(parents=True, exist_ok=True)
201         destino = SALIDA / f"medicion_{args.etiqueta}.json"
202         destino.write_text(
203             json.dumps(resultado, indent=2, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n"
204         )
205         print(json.dumps(resultado, indent=2, ensure_ascii=False))
206         print(f"Escrito {destino.relative_to(RAIZ)}")
207         return 0

211         sys.exit(main())
