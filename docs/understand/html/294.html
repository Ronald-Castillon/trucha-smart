35     def autenticar_y_registrar(c: Contenedor, nombre: str, password: str, request: Request) -> Usuario:
36         """Confirma la transacción también si falla: el contador de intentos y la bitácora persisten."""
37         try:
38             usuario = c.usuarios.autenticar(nombre, password, ip_cliente(request))
39         except ErrorDominio:
40             c.confirmar()
41             raise
42         c.confirmar()
43         return usuario

46     @router.post("/auth/login", response_model=TokenSalida, tags=["autenticación"])
47     def login(
48         datos: LoginEntrada, request: Request, c: Contenedor = Depends(contenedor)
49     ) -> TokenSalida:
50         """Valida credenciales y devuelve un JWT para usar como `Authorization: Bearer`.
51     
52         Tras 3 intentos fallidos seguidos la cuenta queda bloqueada 5 minutos (423, RF-11).
53         """
54         usuario = autenticar_y_registrar(c, datos.nombre_usuario, datos.password, request)
55         tokens = c.recursos.tokens
56         return TokenSalida(
57             access_token=tokens.emitir(usuario),
58             expira_en_segundos=tokens.expira_segundos,
59             usuario=UsuarioSalida.de(usuario),
60         )

86         usuario: Usuario = Depends(usuario_actual),
87         c: Contenedor = Depends(contenedor),
88     ) -> UsuarioSalida:
89         """Registra un usuario con su rol. Solo Administrador."""
90         nuevo = c.usuarios.registrar(

92                 nombre_usuario=datos.nombre_usuario,
93                 nombre_completo=datos.nombre_completo,
94                 password=datos.password,
95                 rol=datos.rol,
96             ),
97             usuario,
98         )
99         c.confirmar()
100         return UsuarioSalida.de(nuevo)
