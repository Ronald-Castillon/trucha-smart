1     """RNF-P: prueba de carga con Locust y medición real de latencia, throughput y memoria.
2     
3     Arranca la API (uvicorn, N workers) sobre una BD con datos demo, ejecuta Locust sin interfaz con
4     `tests/carga/locustfile.py` y muestrea cada segundo la memoria (RSS) y CPU del servidor con psutil.

6     Métricas (fila "Aggregated" de Locust):
7       p95 = percentil 95 del tiempo de respuesta (ms)

9       Errores = peticiones fallidas / peticiones totales x 100
10     
11     Uso:
12       python tools/prueba_carga.py --usuarios 500 --tasa 25 --duracion 180 --workers 4

15     """

43     def puerto_libre() -> int:
44         with socket.socket() as s:

49     class Monitor(threading.Thread):
50         """Muestrea RSS (MB) y CPU (%) del proceso servidor y sus hijos (workers)."""
51     
52         def __init__(self, pid: int) -> None:
53             super().__init__(daemon=True)
54             self.proceso = psutil.Process(pid)
55             self.memoria_mb: list[float] = []
56             self.cpu_pct: list[float] = []
57             self.activo = True
58     
59             self._cache: dict[int, psutil.Process] = {}
60     
61         def _procesos(self):
62             # Se reutilizan los objetos Process: cpu_percent() mide desde la llamada anterior.
63             actuales = [self.proceso, *self.proceso.children(recursive=True)]
64             return [self._cache.setdefault(p.pid, p) for p in actuales]
65     
66         def run(self) -> None:
67             for p in self._procesos():
68                 p.cpu_percent(None)
69             while self.activo:
70                 time.sleep(1)
71                 try:
72                     procs = self._procesos()
73                     self.memoria_mb.append(sum(p.memory_info().rss for p in procs) / 2**20)
74                     self.cpu_pct.append(sum(p.cpu_percent(None) for p in procs))
75                 except psutil.Error:
76                     pass

79     def detener_arbol(servidor: subprocess.Popen) -> None:
80         """Detiene uvicorn y sus workers: en Windows terminate() no alcanza a los procesos hijos,
81         que quedaban huérfanos y consumían CPU en las mediciones siguientes."""
82         try:
83             hijos = psutil.Process(servidor.pid).children(recursive=True)
84         except psutil.Error:
85             hijos = []
86         servidor.terminate()
87         try:
88             servidor.wait(timeout=15)
89         except subprocess.TimeoutExpired:  # conexiones colgadas impiden el cierre ordenado
90             servidor.kill()
91             servidor.wait()
92         for h in hijos:
93             try:
94                 h.kill()
95             except psutil.Error:
96                 pass
97         psutil.wait_procs(hijos, timeout=10)

233     def percentil(valores: list[float], q: float) -> float:
234         """Percentil por rango más cercano (el mismo criterio que usa Locust, sin redondeo)."""
235         orden = sorted(valores)
236         return orden[max(0, math.ceil(q / 100 * len(orden)) - 1)]

239     def fases_de(ruta: Path, rampa_s: float) -> dict:
240         """Separa la rampa de subida (usuarios entrando y autenticándose) del estado estable."""
241         if not ruta.is_file():
242             return {}
243         filas = [(float(t), float(ms), n, int(ok)) for t, ms, n, ok in csv.reader(open(ruta))]
244         if not filas:
245             return {}
246         inicio = min(f[0] for f in filas)
247         resultado = {}
248         for nombre, cond in (
249             ("rampa", lambda t: t - inicio < rampa_s),
250             ("estable", lambda t: t - inicio >= rampa_s),
251         ):
252             sel = [f for f in filas if cond(f[0])]
253             if not sel:
254                 continue
255             ms = [f[1] for f in sel]
256             dur = max(f[0] for f in sel) - min(f[0] for f in sel) or 1
257             resultado[nombre] = {
258                 "segundos": round(dur, 1),
259                 "peticiones": len(sel),
260                 "p50_ms": round(percentil(ms, 50)),
261                 "p95_ms": round(percentil(ms, 95)),
262                 "p99_ms": round(percentil(ms, 99)),
263                 "errores_pct": round(100 * sum(1 for f in sel if not f[3]) / len(sel), 3),
264                 "rps": round(len(sel) / dur, 1),
265             }
266         return resultado
