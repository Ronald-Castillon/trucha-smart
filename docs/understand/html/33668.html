26             self,

37         # -- RegistrarUsuario --------------------------------------------------
38         def registrar(self, comando: ComandoRegistrarUsuario, solicitante: Usuario) -> Usuario:
39             solicitante.exigir(Permiso.GESTIONAR_USUARIOS)
40             nombre = validar_nombre_usuario(comando.nombre_usuario)
41             validar_password(comando.password)
42             nombre_completo = comando.nombre_completo.strip()
43             if not nombre_completo:
44                 raise ErrorValidacion("nombre_completo", "El nombre completo es obligatorio")
45             if self._repo.por_nombre_usuario(nombre) is not None:
46                 raise Conflicto(f"El nombre de usuario '{nombre}' ya está registrado")
47             usuario = Usuario(
48                 nombre_usuario=nombre,
49                 nombre_completo=nombre_completo,
50                 password_hash=self._hasher.hashear(comando.password),
51                 rol=comando.rol,
52                 creado_en=self._reloj.ahora(),
53             )
54             return self._repo.guardar(usuario)
55     
56         def listar(self, solicitante: Usuario) -> list[Usuario]:
57             solicitante.exigir(Permiso.GESTIONAR_USUARIOS)
58             return self._repo.listar()

60         def nombres(self, solicitante: Usuario) -> dict[int, str]:
61             """Nombre completo por id, para mostrar quién registró o atendió algo (B-04, B-06)."""
62             solicitante.exigir(Permiso.CONSULTAR)
63             return {u.id: u.nombre_completo for u in self._repo.listar()}

65         # -- Autenticar --------------------------------------------------------
66         def autenticar(self, nombre_usuario: str, password: str, ip: str | None = None) -> Usuario:
67             """Tras MAXIMO_INTENTOS_FALLIDOS seguidos la cuenta queda bloqueada MINUTOS_BLOQUEO (RF-11).
68     
69             Los intentos fallidos y el bloqueo se guardan aunque se lance la excepción: el adaptador
70             debe confirmar la transacción también en ese caso.
71             """
72             nombre = nombre_usuario.strip().lower()
73             usuario = self._repo.por_nombre_usuario(nombre)
74             ahora = self._reloj.ahora()
75             if usuario is None or not usuario.activo:
76                 self._evento(
77                     TipoEvento.LOGIN_FALLIDO, nombre, None, ip, "usuario inexistente o inactivo"
78                 )
79                 raise CredencialesInvalidas()
80             if usuario.bloqueado(ahora):
81                 self._evento(TipoEvento.LOGIN_RECHAZADO_BLOQUEO, nombre, usuario.id, ip)
82                 raise self._bloqueada(usuario, ahora)
83             if not self._hasher.verificar(password, usuario.password_hash):
84                 se_bloqueo = usuario.registrar_fallo(ahora)
85                 self._repo.guardar(usuario)
86                 self._evento(TipoEvento.LOGIN_FALLIDO, nombre, usuario.id, ip, "contraseña incorrecta")
87                 if se_bloqueo:
88                     self._evento(TipoEvento.CUENTA_BLOQUEADA, nombre, usuario.id, ip)
89                     raise self._bloqueada(usuario, ahora)
90                 raise CredencialesInvalidas()
91             if usuario.intentos_fallidos or usuario.bloqueado_hasta:
92                 usuario.registrar_exito()
93                 usuario = self._repo.guardar(usuario)
94             self._evento(TipoEvento.LOGIN_EXITOSO, nombre, usuario.id, ip)
95             return usuario

97         @staticmethod
98         def _bloqueada(usuario: Usuario, ahora) -> CuentaBloqueada:
99             minutos = math.ceil((usuario.bloqueado_hasta - ahora).total_seconds() / 60)
100             return CuentaBloqueada(
101                 f"Cuenta bloqueada por intentos fallidos. Intente de nuevo en {minutos} min "
102                 "o pida al Administrador que la revise."
103             )
104     
105         def _evento(self, tipo, nombre, usuario_id, ip, detalle: str = "") -> None:
106             if self._auditoria is not None:
107                 self._auditoria.registrar(tipo, nombre, usuario_id, ip, detalle)
108     
109         def usuario_activo(self, usuario_id: int) -> Usuario | None:
110             usuario = self._repo.por_id(usuario_id)
111             return usuario if usuario is not None and usuario.activo else None
