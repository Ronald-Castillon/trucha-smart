51     def ajustar(xs: list[float], ys: list[float]) -> Ajuste:
52         """Mínimos cuadrados y = a + b·x. Con varianza nula en y, R² = 1 si la recta es exacta."""
53         n = len(xs)
54         mx, my = sum(xs) / n, sum(ys) / n
55         sxx = sum((x - mx) ** 2 for x in xs)
56         if sxx == 0:
57             return Ajuste(0.0, my, 0.0)
58         b = sum((x - mx) * (y - my) for x, y in zip(xs, ys, strict=True)) / sxx
59         a = my - b * mx
60         sst = sum((y - my) ** 2 for y in ys)
61         sse = sum((y - (a + b * x)) ** 2 for x, y in zip(xs, ys, strict=True))
62         r2 = 1.0 if sst == 0 else max(0.0, 1 - sse / sst)
63         return Ajuste(b, a, r2)
