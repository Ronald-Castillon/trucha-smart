16     @router.get("/alertas", response_model=list[AlertaSalida])
17     def listar_alertas(
18         estado: EstadoAlerta | None = None,
19         estanque_id: int | None = None,
20         limite: int = Query(100, ge=1, le=500),
21         usuario: Usuario = Depends(usuario_actual),
22         c: Contenedor = Depends(contenedor),
23     ) -> list[AlertaSalida]:
24         """Alertas de la más reciente a la más antigua; filtra por estado y estanque."""
25         return [AlertaSalida.de(a) for a in c.alertas.listar(usuario, estanque_id, estado, limite)]

29     def atender_alerta(
