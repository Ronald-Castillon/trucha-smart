12     import argparse

14     import os
15     from collections import Counter, defaultdict
16     from datetime import UTC, datetime
17     from pathlib import Path
18     
19     CAPAS = ("domain", "ports", "adapters", "main", "tools")
20     # Reglas de pyproject.toml (import-linter, ADR-001): el dominio y los puertos forman el núcleo
21     # y no conocen a los adaptadores; los servicios del dominio usan las interfaces de los puertos.
22     PERMITIDAS = {
23         "domain": {"domain", "ports"},
24         "ports": {"ports", "domain"},
25         "adapters": {"adapters", "ports", "domain"},
26         "main": set(CAPAS),
27         "tools": set(CAPAS),
28     }
29     FUNCIONES = {"Function", "Static Function", "Method", "Public Method", "Private Method"}
30     
31     
32     def numero(valor: str) -> float:
33         """Understand usa la configuración regional (en español, coma decimal)."""
34         valor = (valor or "").strip().replace(",", ".")
35         try:
36             return float(valor)
37         except ValueError:
38             return 0.0
39

41     def capa(ruta: Path, codigo: Path) -> str:
42         partes = ruta.relative_to(codigo).parts
43         if partes[0] == "tools":
44             return "tools"
45         if partes[:2] == ("src", "main.py"):
46             return "main"
47         return partes[1] if len(partes) > 1 and partes[1] in CAPAS else "otro"
48     
49     
50     def mapa_capas(codigo: Path) -> dict[str, str]:
51         """Nombre corto del archivo → capa (solo si el nombre no es ambiguo entre capas)."""
52         candidatos: dict[str, set[str]] = defaultdict(set)
53         for base in ("src", "tools"):
54             for ruta in (codigo / base).rglob("*.py"):
55                 candidatos[ruta.name].add(capa(ruta, codigo))
56         return {n: next(iter(c)) for n, c in candidatos.items() if len(c) == 1}
57     
58     
59     def tabla(cabecera: list[str], filas: list[list[str]]) -> str:
60         lineas = ["| " + " | ".join(cabecera) + " |", "|" + "---|" * len(cabecera)]
61         lineas += ["| " + " | ".join(str(c) for c in f) + " |" for f in filas]
62         return "\n".join(lineas) + "\n"

65     def main(argv: list[str] | None = None) -> int:
66         parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
67         parser.add_argument("--salida", type=Path, required=True)
68         parser.add_argument("--codigo", type=Path, required=True)
69         args = parser.parse_args(argv)
70     
71         filas = list(csv.DictReader((args.salida / "metricas.csv").open(encoding="utf-8-sig")))
72         archivos = [f for f in filas if f["Kind"] == "Module File"]
73         clases = [f for f in filas if f["Kind"] == "Class"]
74         funciones = [f for f in filas if f["Kind"] in FUNCIONES]
75         tipos = Counter(f["Kind"] for f in filas)
76     
77         def suma(lista, clave):
78             return int(sum(numero(f[clave]) for f in lista))
79     
80         loc, comentarios = suma(archivos, "CountLineCode"), suma(archivos, "CountLineComment")
81         ciclo = [numero(f["Cyclomatic"]) for f in funciones]
82         altas = [c for c in ciclo if c > 10]
83     
84         s = ["# Análisis estático con SciTools Understand (guía S14)\n\n"]
85         s.append(
86             f"Generado por `tools/understand/analizar.ps1` en GitHub Actions "
87             f"({datetime.now(UTC):%d/%m/%Y %H:%M} UTC). "
88             f"Corrida: {os.getenv('GITHUB_SERVER_URL', 'https://github.com')}/"
89             f"{os.getenv('GITHUB_REPOSITORY', '')}/actions/runs/{os.getenv('GITHUB_RUN_ID', '')}\n\n"
90         )
91         s.append(
92             "Alcance: `src/` (arquitectura hexagonal) y `tools/`, lenguaje Python. Herramienta: "
93             "SciTools Understand 8.1 con licencia educativa, en el runner propio del equipo.\n"
94         )
95         s.append("\n## 1. Métricas del proyecto\n\n")
96         s.append(tabla(["Métrica", "Valor"], [
97             ["Archivos de código (Module File)", len(archivos)],
98             ["Clases", len(clases)],
99             ["Funciones y métodos", len(funciones)],
100             ["Líneas totales (CountLine)", suma(archivos, "CountLine")],
101             ["Líneas de código (CountLineCode)", loc],
102             ["Líneas de comentario (CountLineComment)", comentarios],
103             ["Relación comentario/código", f"{comentarios / loc:.2f}" if loc else "—"],
104             ["Complejidad ciclomática promedio por función", f"{sum(ciclo) / len(ciclo):.2f}" if ciclo else "—"],
105             ["Complejidad ciclomática máxima", int(max(ciclo, default=0))],
106             ["Funciones con complejidad > 10", f"{len(altas)} de {len(funciones)}"],
107         ]))  # fmt: skip
108         s.append(
109             "\nEntidades exportadas por tipo: "
110             + ", ".join(f"{k} {v}" for k, v in sorted(tipos.items()))
111             + ".\n"
112         )
113     
114         s.append("\n## 2. Métricas por capa (Architecture)\n\n")
115         por_capa: dict[str, list[dict]] = defaultdict(list)
116         nombres = mapa_capas(args.codigo)
117         for f in archivos:
118             por_capa[nombres.get(f["Name"], "otro")].append(f)
119         s.append(tabla(["Capa", "Archivos", "Líneas de código", "Complejidad total (SumCyclomatic)"], [
120             [c, len(por_capa[c]), suma(por_capa[c], "CountLineCode"), suma(por_capa[c], "SumCyclomatic")]
121             for c in (*CAPAS, "otro") if por_capa.get(c)
122         ]))  # fmt: skip
123     
124         s.append("\n## 3. Function Metrics · 10 funciones de mayor complejidad ciclomática\n\n")
125         top = sorted(
126             funciones, key=lambda f: (numero(f["Cyclomatic"]), numero(f["CountLineCode"])), reverse=True
127         )[:10]
128         s.append(tabla(["Función", "Ciclomática", "Líneas de código", "Anidamiento máx."], [
129             [f"`{f['Name']}`", int(numero(f["Cyclomatic"])), int(numero(f["CountLineCode"])), int(numero(f["MaxNesting"]))]
130             for f in top
131         ]))  # fmt: skip
132     
133         s.append("\n## 4. Class Metrics · 10 clases con más acoplamiento\n\n")
134         topc = sorted(
135             clases,
136             key=lambda f: (numero(f["CountClassCoupled"]), numero(f["CountDeclMethod"])),
137             reverse=True,
138         )[:10]
139         s.append(tabla(["Clase", "Acoplamiento (CBO)", "Métodos", "Herencia máx.", "Líneas de código"], [
140             [f"`{f['Name']}`", int(numero(f["CountClassCoupled"])), int(numero(f["CountDeclMethod"])),
141              int(numero(f["MaxInheritanceTree"])), int(numero(f["CountLineCode"]))]
142             for f in topc
143         ]))  # fmt: skip
144     
145         s.append("\n## 5. Dependencies · dependencias entre capas\n\n")
146         matriz: Counter[tuple[str, str]] = Counter()
147         violaciones = []
148         deps = csv.DictReader((args.salida / "dependencias_archivos.csv").open(encoding="utf-8-sig"))
149         for d in deps:
150             origen, destino = nombres.get(d["From File"]), nombres.get(d["To File"])
151             if not origen or not destino:
152                 continue
153             refs = int(numero(d["References"]))
154             matriz[(origen, destino)] += refs
155             if destino not in PERMITIDAS.get(origen, set()):
156                 violaciones.append([d["From File"], origen, d["To File"], destino, refs])
157         usadas = [c for c in CAPAS if any(c in par for par in matriz)]
158         s.append("Referencias desde la capa de la fila hacia la capa de la columna:\n\n")
159         s.append(
160             tabla(
161                 ["Desde \\ Hacia", *usadas],
162                 [[o, *[matriz.get((o, d), 0) or "·" for d in usadas]] for o in usadas],
163             )
164         )
165         s.append(
166             f"\n**Regla de la arquitectura hexagonal (ADR-001, contratos de import-linter en "
167             f"pyproject.toml):** `domain` y `ports` no dependen de `adapters` ni de `main`. "
168             f"Dependencias que la violan según Understand: **{len(violaciones)}**.\n"
169         )
170         if violaciones:
171             s.append(
172                 "\n" + tabla(["Archivo", "Capa", "Depende de", "Capa", "Referencias"], violaciones)
173             )
174     
175         s.append("\n## 6. Archivos del reporte\n\n")
176         s.append(tabla(["Archivo", "Contenido"], [
177             ["[metricas.csv](metricas.csv)", "Todas las métricas por entidad (`und metrics`)"],
178             ["[dependencias_archivos.csv](dependencias_archivos.csv)", "Dependencias entre archivos (`und export -dependencies file csv`)"],
179             ["[dependencias_clases.csv](dependencias_clases.csv)", "Dependencias entre clases (`und export -dependencies class csv`)"],
180             ["[reporte_understand.txt](reporte_understand.txt)", "Reportes Project/File/Class/Class OO/Program Unit Metrics, Complexity e Imports (`und report`)"],
181             ["[html/](html/)", "Los mismos reportes en HTML"],
182             ["[analisis.sarif](analisis.sarif)", "Errores y advertencias del análisis (`und analyze -sarif`)"],
183         ]))  # fmt: skip
184         (args.salida / "README.md").write_text("".join(s), encoding="utf-8", newline="\n")
185         print("".join(s))
186         return 0
