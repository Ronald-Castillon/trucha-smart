1     """Genera db/seed.sql con datos de demostración reproducibles.
2     
3     - 3 usuarios (uno por rol) con contraseña hasheada con bcrypt.
4     - 3 estanques/jaulas con un lote activo cada uno.

7       (una atendida y otra abierta), coherentes con config/umbrales.yaml.
8     - Alimentación diaria y algunos registros de mortalidad.
9     
10     Uso:  python tools/generar_seed.py
11     Las contraseñas de demostración están documentadas en el README.
12     """

14     from __future__ import annotations

16     import math
17     import random
18     import sys

28     from src.adapters.config import cargar_umbrales  # noqa: E402
29     from src.domain.model.umbral import Nivel  # noqa: E402
30     
31     # Sal fija para que el archivo generado sea reproducible (solo datos de demostración).
32     SAL_DEMO = b"$2b$12$TruchaSmartDemoSaltSee"
33     
34     USUARIOS = [
35         (1, "admin", "Ana Administradora", "Admin123!", "ADMINISTRADOR"),
36         (2, "piscicultor", "Pedro Piscicultor", "Pisci123!", "PISCICULTOR"),
37         (3, "operario", "Olga Operaria", "Oper123!", "OPERARIO"),
38     ]
39     ESTANQUES = [
40         (1, "Estanque A1", "ESTANQUE", 120.0, "Sector norte"),
41         (2, "Estanque A2", "ESTANQUE", 120.0, "Sector norte"),
42         (3, "Jaula J1", "JAULA", 250.0, "Laguna, muelle 2"),
43     ]
44     # id, estanque, código, peces, peso inicial, peso actual, siembra
45     LOTES = [
46         (1, 1, "L-2026-01", 5000, 45.0, 62.0, "2026-08-01"),
47         (2, 2, "L-2026-02", 8000, 8.0, 15.0, "2026-09-10"),
48         (3, 3, "L-2026-03", 3000, 180.0, 230.0, "2026-06-15"),
49     ]
50     INICIO = datetime(2026, 10, 16, 0, 0, tzinfo=UTC)
51     DIAS = 7
52     PASO_HORAS = 3
53     
54     # Ventanas (inicio, fin) en las que la jaula J1 tiene OD en precaución (5.0 <= OD < 7.0).
55     CAIDAS_OD_J1 = [
56         (INICIO + timedelta(days=2, hours=3), INICIO + timedelta(days=2, hours=9)),
57         (INICIO + timedelta(days=6, hours=12), INICIO + timedelta(days=6, hours=21)),
58     ]
59     
60     
61     def ts(d: datetime) -> str:
62         return d.strftime("'%Y-%m-%d %H:%M:%S+00'")
63     
64     
65     def generar() -> str:
66         rnd = random.Random(20261023)
67         umbral_od = cargar_umbrales(RAIZ / "config" / "umbrales.yaml").oxigeno_disuelto
68         out: list[str] = [
69             "-- Generado por tools/generar_seed.py. No editar a mano: regenerar.",
70             "-- Datos de demostración (contraseñas en el README).",
71             "",
72         ]
73     
74         out.append(
75             "INSERT INTO usuarios (id, nombre_usuario, nombre_completo, password_hash, rol, activo, "
76             "creado_en) VALUES"
77         )
78         filas = []
79         for uid, user, nombre, pwd, rol in USUARIOS:
80             h = bcrypt.hashpw(pwd.encode(), SAL_DEMO).decode()
81             filas.append(f"  ({uid}, '{user}', '{nombre}', '{h}', '{rol}', TRUE, {ts(INICIO)})")
82         out.append(",\n".join(filas) + ";\n")
83     
84         out.append(
85             "INSERT INTO estanques (id, nombre, tipo, volumen_m3, ubicacion, activo, creado_en) VALUES"
86         )
87         out.append(
88             ",\n".join(
89                 f"  ({e[0]}, '{e[1]}', '{e[2]}', {e[3]:.2f}, '{e[4]}', TRUE, {ts(INICIO)})"
90                 for e in ESTANQUES
91             )
92             + ";\n"
93         )
94     
95         out.append(
96             "INSERT INTO lotes (id, estanque_id, codigo, num_peces_inicial, peso_medio_inicial_g, "
97             "peso_medio_actual_g, fecha_siembra, activo, fecha_cierre) VALUES"
98         )
99         out.append(
100             ",\n".join(
101                 f"  ({lo[0]}, {lo[1]}, '{lo[2]}', {lo[3]}, {lo[4]:.2f}, {lo[5]:.2f}, '{lo[6]}', "
102                 "TRUE, NULL)"
103                 for lo in LOTES
104             )
105             + ";\n"
106         )
107     
108         lecturas = []
109         alertas = []
110         lid = 0
111         for est_id, *_ in ESTANQUES:
112             alerta_abierta: dict | None = None
113             t = INICIO
114             while t < INICIO + timedelta(days=DIAS):
115                 lid += 1
116                 hora = t.hour + t.minute / 60
117                 temp = 12.5 + 1.8 * math.sin((hora - 9) / 24 * 2 * math.pi) + rnd.uniform(-0.3, 0.3)
118                 od = 8.4 - 0.5 * math.sin((hora - 9) / 24 * 2 * math.pi) + rnd.uniform(-0.3, 0.3)
119                 ph = 7.3 + rnd.uniform(-0.25, 0.25)
120                 en_caida = est_id == 3 and any(a <= t <= b for a, b in CAIDAS_OD_J1)
121                 if en_caida:
122                     od = rnd.uniform(5.6, 6.6)
123                 fuente = "SENSOR" if est_id != 2 else "MANUAL"
124                 reg = "NULL" if fuente == "SENSOR" else "3"
125                 lecturas.append(
126                     f"  ({lid}, {est_id}, {temp:.2f}, {od:.2f}, {ph:.2f}, '{fuente}', {reg}, {ts(t)})"
127                 )
128                 if en_caida and alerta_abierta is None:
129                     alerta_abierta = {
130                         "id": len(alertas) + 1,
131                         "lectura": lid,
132                         "valor": od,
133                         "creada": t,
134                     }
135                     alertas.append(alerta_abierta)
136                 if not en_caida and alerta_abierta is not None:
137                     # La primera caída se atendió 1 h después de crearse la alerta.
138                     alerta_abierta["atendida"] = alerta_abierta["creada"] + timedelta(hours=1)
139                     alerta_abierta = None
140                 t += timedelta(hours=PASO_HORAS)
141     
142         out.append(
143             "INSERT INTO lecturas_agua (id, estanque_id, temperatura_c, oxigeno_mg_l, ph, fuente, "
144             "registrado_por, medido_en) VALUES"
145         )
146         out.append(",\n".join(lecturas) + ";\n")
147     
148         out.append(
149             "INSERT INTO alertas (id, estanque_id, lectura_id, parametro, nivel, valor, mensaje, "
150             "estado, creada_en, atendida_por, atendida_en, observacion) VALUES"
151         )
152         filas = []
153         for a in alertas:
154             # Mismo formato que las alertas generadas por el sistema (Issue #11).
155             msg = umbral_od.describir(round(a["valor"], 2), Nivel.PRECAUCION)
156             if "atendida" in a:
157                 filas.append(
158                     f"  ({a['id']}, 3, {a['lectura']}, 'OXIGENO_DISUELTO', 'PRECAUCION', "
159                     f"{a['valor']:.2f}, '{msg}', 'ATENDIDA', {ts(a['creada'])}, 3, "
160                     f"{ts(a['atendida'])}, 'Se encendió el aireador')"
161                 )
162             else:
163                 filas.append(
164                     f"  ({a['id']}, 3, {a['lectura']}, 'OXIGENO_DISUELTO', 'PRECAUCION', "
165                     f"{a['valor']:.2f}, '{msg}', 'ABIERTA', {ts(a['creada'])}, NULL, NULL, NULL)"
166                 )
167         out.append(",\n".join(filas) + ";\n")
168     
169         alim = []
170         aid = 0
171         tasas = {1: 0.02, 2: 0.03, 3: 0.012}  # según config/alimentacion.yaml
172         for lote_id, _, _, peces, _, peso_act, _ in LOTES:
173             for d in range(DIAS):
174                 aid += 1
175                 kg = peces * peso_act / 1000 * tasas[lote_id] * rnd.uniform(0.95, 1.05)
176                 momento = INICIO + timedelta(days=d, hours=8)
177                 alim.append(f"  ({aid}, {lote_id}, {kg:.3f}, 3, {ts(momento)})")
178         out.append(
179             "INSERT INTO registros_alimentacion (id, lote_id, cantidad_kg, registrado_por, "

182         out.append(",\n".join(alim) + ";\n")
183     
184         mort = [
185             f"  (1, 1, 12, 'Manipulación en biometría', 3, {ts(INICIO + timedelta(days=1, hours=17))})",
186             f"  (2, 2, 25, 'Desconocida', 3, {ts(INICIO + timedelta(days=3, hours=17))})",
187             f"  (3, 3, 8, 'Estrés por bajo oxígeno', 3, {ts(INICIO + timedelta(days=2, hours=17))})",
188         ]
189         out.append(
190             "INSERT INTO registros_mortalidad (id, lote_id, cantidad, causa, registrado_por, "
191             "registrado_en) VALUES"
192         )
193         out.append(",\n".join(mort) + ";\n")
194     
195         # Reajustar secuencias (solo PostgreSQL; tools/init_db.py omite estas líneas en SQLite).
196         out.append("-- postgresql-only")
197         for tabla in (
198             "usuarios",
199             "estanques",
200             "lotes",
201             "lecturas_agua",
202             "alertas",
203             "registros_alimentacion",
204             "registros_mortalidad",
205         ):
206             out.append(
207                 f"SELECT setval(pg_get_serial_sequence('{tabla}', 'id'), "
208                 f"(SELECT MAX(id) FROM {tabla}));"
209             )
210         return "\n".join(out) + "\n"
