10     import threading
11     from collections.abc import Callable
12     from dataclasses import dataclass, field
13     from datetime import UTC, datetime
14     
15     from src.adapters.inbound.simulator.escenarios import ESCENARIOS, Medicion, generar
16     
17     INTERVALO_MINIMO_S = 1.0
18     INTERVALO_MAXIMO_S = 60.0
19     MAXIMO_SIMULACIONES = 10
20     
21     # (estanque_id, usuario_id, medición) -> None. Lanza una excepción si la lectura no se registra.
22     EjecutarPaso = Callable[[int, int, Medicion], None]
23     
24     
25     class ErrorSimulacion(ValueError):
26         """Parámetros de simulación inválidos o límite alcanzado."""
27     
28     
29     @dataclass
30     class EstadoSimulacion:
31         estanque_id: int
32         escenario: str
33         intervalo_s: float
34         usuario_id: int
35         iniciada_en: datetime
36         lecturas_enviadas: int = 0
37         ultimo_error: str | None = None
38         _detener: threading.Event = field(default_factory=threading.Event, repr=False)
39         _hilo: threading.Thread | None = field(default=None, repr=False)
40     
41         @property
42         def activa(self) -> bool:
43             return self._hilo is not None and self._hilo.is_alive() and not self._detener.is_set()
44     
45     
46     class SimuladorEnProceso:
47         def __init__(self, ejecutar_paso: EjecutarPaso) -> None:
48             self._ejecutar_paso = ejecutar_paso
49             self._simulaciones: dict[int, EstadoSimulacion] = {}
50             self._candado = threading.Lock()
51     
52         def iniciar(
53             self, estanque_id: int, escenario: str, intervalo_s: float, usuario_id: int
54         ) -> EstadoSimulacion:
55             if escenario not in ESCENARIOS:
56                 raise ErrorSimulacion(f"Escenario desconocido. Opciones: {', '.join(ESCENARIOS)}")
57             if not INTERVALO_MINIMO_S <= intervalo_s <= INTERVALO_MAXIMO_S:
58                 raise ErrorSimulacion(
59                     f"El intervalo debe estar entre {INTERVALO_MINIMO_S:g} y {INTERVALO_MAXIMO_S:g} s"
60                 )
61             with self._candado:
62                 self._detener_sin_candado(estanque_id)
63                 if len(self.activas()) >= MAXIMO_SIMULACIONES:
64                     raise ErrorSimulacion(f"Máximo {MAXIMO_SIMULACIONES} simulaciones a la vez")
65                 estado = EstadoSimulacion(
66                     estanque_id, escenario, intervalo_s, usuario_id, datetime.now(UTC)
67                 )
68                 estado._hilo = threading.Thread(
69                     target=self._bucle, args=(estado,), daemon=True, name=f"sim-estanque-{estanque_id}"
70                 )
71                 self._simulaciones[estanque_id] = estado
72                 estado._hilo.start()
73                 return estado
74     
75         def detener(self, estanque_id: int) -> bool:
76             with self._candado:
77                 return self._detener_sin_candado(estanque_id)
78     
79         def detener_todas(self) -> None:
80             with self._candado:
81                 for estanque_id in list(self._simulaciones):
82                     self._detener_sin_candado(estanque_id)
83     
84         def activas(self) -> list[EstadoSimulacion]:
85             return [s for s in self._simulaciones.values() if s.activa]
86     
87         def de(self, estanque_id: int) -> EstadoSimulacion | None:
88             estado = self._simulaciones.get(estanque_id)
89             return estado if estado is not None and estado.activa else None
90     
91         def _detener_sin_candado(self, estanque_id: int) -> bool:
92             estado = self._simulaciones.pop(estanque_id, None)
93             if estado is None:
94                 return False
95             estado._detener.set()
96             if estado._hilo is not None and estado._hilo is not threading.current_thread():
97                 estado._hilo.join(timeout=5)
98             return True
99     
100         def _bucle(self, estado: EstadoSimulacion) -> None:
101             for medicion in generar(estado.escenario):
102                 if estado._detener.is_set():
103                     return
104                 try:
105                     self._ejecutar_paso(estado.estanque_id, estado.usuario_id, medicion)
106                     estado.lecturas_enviadas += 1
107                     estado.ultimo_error = None
108                 except Exception as e:  # noqa: BLE001 - se informa y se detiene la simulación
109                     estado.ultimo_error = str(getattr(e, "mensaje", e))
110                     estado._detener.set()
111                     return
112                 if estado._detener.wait(estado.intervalo_s):
113                     return
