64     def _pendiente(puntos: list[tuple[datetime, float]], fin: datetime) -> float:
65         recientes = [(t, v) for t, v in puntos if t > fin - VENTANA_TENDENCIA]
66         if len(recientes) < 2:
67             return 0.0
68         xs = [(t - fin).total_seconds() / 3600 for t, _ in recientes]
69         return ajustar(xs, [v for _, v in recientes]).pendiente_por_hora

71     
72     def caracteristicas(
73         serie: list[tuple[datetime, float, float, float]], od_limite: float, t_limite: float
74     ) -> dict[str, float] | None:
75         """`serie`: (momento, temperatura, oxígeno, pH) de la ventana de 24 h, en cualquier orden.
76     
77         Devuelve None si hay menos de MINIMO_PUNTOS lecturas.
78         """
79         if len(serie) < MINIMO_PUNTOS:
80             return None
81         serie = sorted(serie)
82         fin, t_act, od_act, ph_act = serie[-1]
83         return {
84             "od_actual": od_act,
85             "t_actual": t_act,
86             "ph_actual": ph_act,
87             "od_pendiente_6h": _pendiente([(m, od) for m, _, od, _ in serie], fin),
88             "t_pendiente_6h": _pendiente([(m, t) for m, t, _, _ in serie], fin),
89             "od_min_24h": min(od for _, _, od, _ in serie),
90             "t_max_24h": max(t for _, t, _, _ in serie),
91             "od_margen": od_act - od_limite,
92             "t_margen": t_limite - t_act,
93         }
94

96     def nivel_de(probabilidad: float) -> NivelRiesgoModelo:
97         if probabilidad >= UMBRAL_ALTO:
98             return NivelRiesgoModelo.ALTO
99         if probabilidad >= UMBRAL_MEDIO:
100             return NivelRiesgoModelo.MEDIO
101         return NivelRiesgoModelo.BAJO

104     class ServicioRiesgoModelo:
105         """Implementa el puerto de entrada EvaluarRiesgoModelo."""
106     
107         def __init__(
108             self,
109             lecturas: RepositorioLectura,
110             estanques: RepositorioEstanque,
111             umbrales: ConjuntoUmbrales,
112             modelo: ModeloRiesgo,
113         ) -> None:
114             self._lecturas = lecturas
115             self._estanques = estanques
116             self._umbrales = umbrales
117             self._modelo = modelo
118     
119         def evaluar(self, estanque_id: int, solicitante: Usuario) -> EvaluacionRiesgoModelo:
120             solicitante.exigir(Permiso.CONSULTAR)
121             estanque = self._estanques.por_id(estanque_id)
122             if estanque is None:
123                 raise NoEncontrado(f"No existe el estanque {estanque_id}")
124             ultima = self._lecturas.ultima_de(estanque_id)
125             lecturas: list[LecturaAgua] = (
126                 self._lecturas.de_estanque(estanque_id, ultima.medido_en - VENTANA, ultima.medido_en)
127                 if ultima
128                 else []
129             )
130             od_lim = self._umbrales.oxigeno_disuelto.precaucion_min
131             t_lim = self._umbrales.temperatura.precaucion_max
132             x = caracteristicas(
133                 [(x.medido_en, x.temperatura_c, x.oxigeno_mg_l, x.ph) for x in lecturas], od_lim, t_lim
134             )
135             base = dict(estanque=estanque, version_modelo=self._modelo.version, puntos=len(lecturas))
136             if x is None:
137                 return EvaluacionRiesgoModelo(
138                     **base,
139                     nivel=NivelRiesgoModelo.DATOS_INSUFICIENTES,
140                     probabilidad=None,
141                     factores=[],
142                     basada_en_hasta=ultima.medido_en if ultima else None,
143                     mensaje=f"Datos insuficientes: {len(lecturas)} lecturas en 24 h "
144                     f"(se necesitan {MINIMO_PUNTOS})",
145                 )
146             if x["od_margen"] < 0 or x["t_margen"] < 0:
147                 return EvaluacionRiesgoModelo(
148                     **base,
149                     nivel=NivelRiesgoModelo.YA_CRITICO,
150                     probabilidad=1.0,
151                     factores=[],
152                     basada_en_hasta=ultima.medido_en,
153                     mensaje="El estanque ya está en condición crítica: atender la alerta",
154                 )
155             r = self._modelo.predecir(x)
156             factores = sorted(
157                 (Factor(NOMBRES[k], round(v, 3)) for k, v in r.aportes.items() if v > 0),
158                 key=lambda f: -f.aporte,
159             )[:3]
160             nivel = nivel_de(r.probabilidad)
161             return EvaluacionRiesgoModelo(
162                 **base,
163                 nivel=nivel,
164                 probabilidad=round(r.probabilidad, 4),
165                 factores=factores,
166                 basada_en_hasta=ultima.medido_en,
167                 mensaje=(
168                     f"Probabilidad de evento crítico en {HORIZONTE_HORAS} h: "
169                     f"{r.probabilidad:.0%} ({nivel.value})"
170                 ),
171             )
