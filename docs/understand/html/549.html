1     """Clave de integración para sistemas externos (RF-09).
2     
3     La clave completa solo se conoce al crearla; se guarda su huella SHA-256 y un prefijo visible
4     para identificarla. Generación con `secrets` (aleatoriedad criptográfica, OWASP A02).
5     """
6     
7     from __future__ import annotations
8     
9     import hashlib
10     import hmac
11     import secrets
12     from dataclasses import dataclass
13     from datetime import datetime
14     
15     from src.domain.errores import ErrorValidacion
16     
17     PREFIJO_CLAVE = "ts_"
18     LARGO_PREFIJO_VISIBLE = 11
19     
20     
21     def huella(clave: str) -> str:
22         return hashlib.sha256(clave.encode("utf-8")).hexdigest()
23     
24     
25     def generar_clave() -> str:
26         return PREFIJO_CLAVE + secrets.token_urlsafe(32)
27     
28     
29     @dataclass
30     class ClaveIntegracion:
31         nombre: str
32         prefijo: str
33         hash_clave: str
34         creada_en: datetime
35         creada_por: int | None = None
36         activa: bool = True
37         ultimo_uso: datetime | None = None
38         id: int | None = None
39     
40         def __post_init__(self) -> None:
41             self.nombre = self.nombre.strip()
42             if not (3 <= len(self.nombre) <= 80):
43                 raise ErrorValidacion(
44                     "nombre", "El nombre de la integración debe tener 3 a 80 caracteres"
45                 )
46     
47         @classmethod
48         def nueva(
49             cls, nombre: str, creada_por: int | None, momento: datetime
50         ) -> tuple[ClaveIntegracion, str]:
51             """Devuelve la entidad y la clave en claro (que no se vuelve a mostrar)."""
52             clave = generar_clave()
53             entidad = cls(
54                 nombre=nombre,
55                 prefijo=clave[:LARGO_PREFIJO_VISIBLE],
56                 hash_clave=huella(clave),
57                 creada_en=momento,
58                 creada_por=creada_por,
59             )
60             return entidad, clave
61     
62         def corresponde(self, clave: str) -> bool:
63             return self.activa and hmac.compare_digest(self.hash_clave, huella(clave))
64     
65         def revocar(self) -> None:
66             self.activa = False
