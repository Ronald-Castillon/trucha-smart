1     """RNF-R: prueba de confiabilidad por inyección de fallos (caos) con medición real.
2     
3     1. Un *supervisor* arranca la API con uvicorn y la reinicia apenas termina (como la política
4        `restart: always` de docker-compose o `Restart=always` de systemd).
5     2. Un *inyector de fallos* mata el proceso del servidor (`kill`) N veces durante la prueba.
6     3. Una *sonda* consulta `GET /api/salud` cada 250 ms durante toda la prueba.
7     4. Opcional (`--pg-contenedor`): detiene y vuelve a iniciar el contenedor de PostgreSQL y mide si
8        la API sigue respondiendo (503) durante la caída y cuánto tarda en volver a 200 (reconexión).
9     
10     Métricas (sobre las muestras de la sonda):
11       Disponibilidad = (duración - tiempo caído) / duración x 100, con tiempo caído = suma de los
12         tiempos de recuperación (por tiempo y no por número de sondas: en Windows una sonda a un
13         puerto cerrado tarda 1-2 s en fallar, y contar sondas subestimaría la caída)
14       MTTR = promedio de (primer 200 tras el fallo - instante del fallo)
15       MTBF = tiempo en servicio / número de fallos
16     
17     Uso:
18       python tools/prueba_confiabilidad.py --duracion 300 --fallos 5
19       python tools/prueba_confiabilidad.py --url-bd postgresql+psycopg://... --pg-contenedor ID
20     Guarda docs/calidad/confiabilidad.json y .md (o --salida).
21     """
22

47     def puerto_libre() -> int:
48         with socket.socket() as s:
49             s.bind(("127.0.0.1", 0))
50             return s.getsockname()[1]

53     class Supervisor(threading.Thread):
54         """Mantiene el servidor vivo: si el proceso termina, lo vuelve a lanzar."""
55     
56         def __init__(self, puerto: int, env: dict) -> None:
57             super().__init__(daemon=True)
58             self.puerto, self.env = puerto, env
59             self.proceso: subprocess.Popen | None = None
60             self.reinicios = 0
61             self.activo = True
62             self._listo = threading.Event()
63     
64         def _lanzar(self) -> None:
65             self.proceso = subprocess.Popen(
66                 [sys.executable, "-m", "uvicorn", "src.main:app", "--port", str(self.puerto)],
67                 cwd=RAIZ,
68                 env=self.env,
69                 stdout=subprocess.DEVNULL,
70                 stderr=subprocess.DEVNULL,
71             )
72     
73         def run(self) -> None:
74             self._lanzar()
75             self._listo.set()
76             while self.activo:
77                 if self.proceso.poll() is not None and self.activo:
78                     self.reinicios += 1
79                     self._lanzar()
80                 time.sleep(0.05)
81     
82         def matar(self) -> None:
83             self._listo.wait()
84             self.proceso.kill()
85     
86         def detener(self) -> None:
87             self.activo = False
88             if self.proceso and self.proceso.poll() is None:
89                 self.proceso.terminate()
90                 try:
91                     self.proceso.wait(timeout=10)
92                 except subprocess.TimeoutExpired:
93                     self.proceso.kill()

96     class Sonda(threading.Thread):
97         def __init__(self, url: str, intervalo: float) -> None:
98             super().__init__(daemon=True)
99             self.url, self.intervalo = url, intervalo
100             # (instante de la respuesta, código HTTP o 0 si no conecta). Se usa el instante en que llega
101             # la respuesta: en Windows la conexión al puerto caído espera a que el servidor reinicie.
102             self.muestras: list[tuple[float, int]] = []
103             self.activo = True
104     
105         def run(self) -> None:
106             with httpx.Client(timeout=4.0) as cliente:  # > connect_timeout de la BD (3 s)
107                 while self.activo:
108                     t = time.monotonic()
109                     try:
110                         codigo = cliente.get(f"{self.url}/api/salud").status_code
111                     except httpx.HTTPError:
112                         codigo = 0
113                     self.muestras.append((time.monotonic(), codigo))
114                     time.sleep(max(0.0, self.intervalo - (time.monotonic() - t)))
115

117     def esperar_200(url: str, limite_s: float = 60) -> None:
118         fin = time.monotonic() + limite_s
119         while time.monotonic() < fin:
120             try:
121                 if httpx.get(f"{url}/api/salud", timeout=0.5).status_code == 200:
122                     return
123             except httpx.HTTPError:
124                 pass
125             time.sleep(0.1)
126         raise SystemExit("La API no respondió 200 al iniciar la prueba")

129     def recuperacion(muestras, instante: float) -> float | None:
130         """Segundos desde `instante` hasta la primera sonda con 200 posterior."""
131         return next((t - instante for t, c in muestras if t > instante and c == 200), None)

134     def main(argv: list[str] | None = None) -> int:
135         ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawTextHelpFormatter)
136         ap.add_argument("--duracion", type=float, default=300, help="segundos de prueba (def. 300)")
137         ap.add_argument("--fallos", type=int, default=5, help="veces que se mata el servidor")
138         ap.add_argument("--intervalo", type=float, default=0.25, help="segundos entre sondas")
139         ap.add_argument("--url-bd", help="DATABASE_URL (por defecto, SQLite temporal con datos demo)")
140         ap.add_argument("--pg-contenedor", help="contenedor Docker de PostgreSQL para la caída de BD")
141         ap.add_argument("--pg-caida", type=float, default=10, help="segundos con PostgreSQL detenido")
142         ap.add_argument("--salida", default=str(RAIZ / "docs/calidad/confiabilidad"))
143         ap.add_argument("--semilla", type=int, default=10)
144         args = ap.parse_args(argv)
145         sys.stdout.reconfigure(encoding="utf-8")
146         random.seed(args.semilla)
147     
148         url_bd = args.url_bd or f"sqlite:///{Path(tempfile.mkdtemp()).as_posix()}/caos.db"
149         env = {
150             **os.environ,
151             "DATABASE_URL": url_bd,
152             "JWT_SECRET": os.environ.get("JWT_SECRET", "clave-de-prueba-de-confiabilidad-32b!"),
153             "PYTHONIOENCODING": "utf-8",
154         }
155         subprocess.run(
156             [sys.executable, "tools/init_db.py", "--url", url_bd], cwd=RAIZ, env=env, check=True
157         )
158         puerto = puerto_libre()
159         url = f"http://127.0.0.1:{puerto}"
160         supervisor = Supervisor(puerto, env)
161         supervisor.start()
162         esperar_200(url)
163     
164         sonda = Sonda(url, args.intervalo)
165         inicio = time.monotonic()
166         sonda.start()
167         # Fallos repartidos en la ventana central, con separación mínima para medir cada recuperación.
168         tramo = args.duracion / (args.fallos + 1)
169         momentos = [tramo * (i + 1) + random.uniform(-tramo / 4, tramo / 4) for i in range(args.fallos)]
170         fallos: list[float] = []
171         for m in momentos:
172             time.sleep(max(0.0, inicio + m - time.monotonic()))
173             fallos.append(time.monotonic())
174             supervisor.matar()
175             print(f"  fallo inyectado en t={fallos[-1] - inicio:6.1f} s (kill del servidor)")
176     
177         caida_bd = None
178         if args.pg_contenedor:
179             time.sleep(5)
180             t0 = time.monotonic()
181             subprocess.run(["docker", "stop", args.pg_contenedor], check=True, capture_output=True)
182             print(f"  PostgreSQL detenido en t={t0 - inicio:6.1f} s")
183             time.sleep(args.pg_caida)
184             t_inicio_pg = time.monotonic()
185             subprocess.run(["docker", "start", args.pg_contenedor], check=True, capture_output=True)
186             durante = [c for t, c in sonda.muestras if t0 + 1 < t < t_inicio_pg]
187             espera = recuperacion_bloqueante(sonda, t_inicio_pg)
188             caida_bd = {
189                 "segundos_detenido": round(t_inicio_pg - t0, 1),
190                 "respuestas_durante_caida": {str(c): durante.count(c) for c in sorted(set(durante))},
191                 "api_siguio_respondiendo": bool(durante) and all(c == 503 for c in durante),
192                 "reconexion_s_desde_docker_start": espera,
193             }
194     
195         time.sleep(max(0.0, inicio + args.duracion - time.monotonic()))
196         sonda.activo = False
197         sonda.join()
198         supervisor.detener()
199     
200         muestras = sonda.muestras
201         total = len(muestras)
202         ok = sum(1 for _, c in muestras if c == 200)
203         tr = [recuperacion(muestras, f) for f in fallos]
204         tr_validos = [x for x in tr if x is not None]
205         duracion = muestras[-1][0] - muestras[0][0]
206         caido = sum(tr_validos)
207         resultado = {
208             "ejecutado_en": datetime.now(UTC).isoformat(timespec="seconds"),
209             "entorno": {
210                 "so": f"{platform.system()} {platform.release()}",
211                 "python": platform.python_version(),
212                 "base_de_datos": url_bd.split(":")[0],
213                 "nucleos_cpu": os.cpu_count(),
214             },
215             "parametros": {
216                 "duracion_s": args.duracion,
217                 "fallos_inyectados": len(fallos),
218                 "intervalo_sonda_s": args.intervalo,
219             },
220             "sondas": {"total": total, "con_200": ok, "sin_respuesta_o_error": total - ok},
221             "disponibilidad_pct": round(100 * (duracion - caido) / duracion, 3),
222             "sondas_200_pct": round(100 * ok / total, 3),
223             "tiempos_recuperacion_s": [round(x, 2) if x is not None else None for x in tr],
224             "mttr_s": round(statistics.mean(tr_validos), 2) if tr_validos else None,
225             "mtbf_s": round((duracion - caido) / len(fallos), 1) if fallos else None,
226             "reinicios_del_supervisor": supervisor.reinicios,
227             "caida_postgresql": caida_bd,
228             "metas": {"disponibilidad_pct": DISPONIBILIDAD_META, "mttr_s": MTTR_META_S},
229         }
230         resultado["fallos_tolerables_dia"] = (
231             int(86400 * (1 - DISPONIBILIDAD_META / 100) / resultado["mttr_s"])
232             if resultado["mttr_s"]
233             else None
234         )
235         resultado["cumple"] = {
236             "disponibilidad": resultado["disponibilidad_pct"] >= DISPONIBILIDAD_META,
237             "mttr": resultado["mttr_s"] is not None and resultado["mttr_s"] < MTTR_META_S,
238         }
239         guardar(resultado, Path(args.salida))
240         print(json.dumps(resultado, ensure_ascii=False, indent=2))
241         return 0

244     def recuperacion_bloqueante(sonda: Sonda, desde: float, limite_s: float = 120) -> float | None:
245         fin = time.monotonic() + limite_s
246         while time.monotonic() < fin:
247             r = recuperacion(list(sonda.muestras), desde)
248             if r is not None:
249                 return round(r, 2)
250             time.sleep(0.2)
251         return None

300             "A = MTBF / (MTBF + MTTR); con el MTTR medido, la meta de 99.5 % (432 s de caída al día)",
301             f"se mantiene con hasta **{r['fallos_tolerables_dia']} caídas del proceso por día**.",
302             "",
303             "**Alcance:** recuperación ante caídas del proceso y de la BD en un servidor con",
