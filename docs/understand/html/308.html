52     def _json(estado: int, error: str, detalle: str, campo: str | None = None) -> JSONResponse:
53         return JSONResponse(
54             status_code=estado, content={"error": error, "detalle": detalle, "campo": campo}
55         )

58     def registrar_manejadores(app: FastAPI) -> None:
59         @app.exception_handler(ErrorDominio)
60         async def _dominio(request: Request, exc: ErrorDominio):
61             estado, codigo = next(
62                 ((e, c) for tipo, e, c in _MAPA if isinstance(exc, tipo)), (400, "error_dominio")
63             )
64             if not request.url.path.startswith("/api"):
65                 from src.adapters.inbound.web.plantillas import render
66     
67                 return render(
68                     request,
69                     "error.html",
70                     estado,
71                     titulo_error="No se pudo completar la acción",
72                     mensaje=exc.mensaje,
73                 )
74             return _json(estado, codigo, exc.mensaje, getattr(exc, "campo", None))
75     
76         @app.exception_handler(NoAutenticado)
77         async def _no_autenticado(request: Request, exc: NoAutenticado):
78             if request.url.path.startswith("/api"):
79                 return _json(401, "no_autenticado", "Debe iniciar sesión")
80             return RedirectResponse("/login", status_code=303)
81     
82         @app.exception_handler(RequestValidationError)
83         async def _validacion(request: Request, exc: RequestValidationError):
84             errores = exc.errors()
85             primero = errores[0] if errores else {}
86             loc = [str(p) for p in primero.get("loc", ()) if p not in ("body", "query", "path")]
87             campo = ".".join(loc) or None
88             detalle = f"{campo}: {primero.get('msg', 'dato inválido')}" if campo else "Datos inválidos"
89             return _json(422, "validacion", detalle, campo)
