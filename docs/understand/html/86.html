24     # Límites físicos: fuera de ellos el dato es imposible (error de sensor o de digitación),
25     # no una condición del agua. No confundir con los umbrales de calidad (config/umbrales.yaml).
26     LIMITES_FISICOS: dict[Parametro, tuple[float, float]] = {
27         Parametro.TEMPERATURA: (0.0, 40.0),
28         Parametro.OXIGENO_DISUELTO: (0.0, math.inf),
29         Parametro.PH: (0.0, 14.0),
30     }
31     CAMPOS = {
32         Parametro.TEMPERATURA: ("temperatura_c", "La temperatura", "°C"),
33         Parametro.OXIGENO_DISUELTO: ("oxigeno_mg_l", "El oxígeno disuelto", "mg/L"),
34         Parametro.PH: ("ph", "El pH", ""),
35     }
36     
37     
38     def validar_fisicamente(parametro: Parametro, valor: float) -> None:
39         campo, nombre, unidad = CAMPOS[parametro]
40         minimo, maximo = LIMITES_FISICOS[parametro]
41         if valor is None or not isinstance(valor, int | float) or math.isnan(valor):
42             raise ErrorValidacion(campo, f"{nombre} es obligatorio y debe ser numérico")
43         if not (minimo <= valor <= maximo):
44             rango = f">= {minimo:g}" if maximo == math.inf else f"entre {minimo:g} y {maximo:g}"
45             raise ErrorValidacion(
46                 campo,
47                 f"{nombre} {valor:g} {unidad} es físicamente imposible (debe estar {rango})".replace(
48                     "  ", " "
49                 ),
50             )
