216         return request.app.state.recursos

219     def contenedor(request: Request) -> Iterator[Contenedor]:
220         r: Recursos = request.app.state.recursos
221         sesion = r.bd.sesion()
222         try:
223             yield Contenedor(sesion, r)
224         except Exception:
225             sesion.rollback()
226             raise
227         finally:
228             sesion.close()

234     def usuario_actual(
235         request: Request,
236         credenciales: HTTPAuthorizationCredentials | None = Depends(_bearer),
237         c: Contenedor = Depends(contenedor),
238     ) -> Usuario:
239         token = credenciales.credentials if credenciales else request.cookies.get(COOKIE_SESION)
240         usuario_id = c.recursos.tokens.usuario_id(token) if token else None
241         usuario = c.usuarios.usuario_activo(usuario_id) if usuario_id is not None else None
242         if usuario is None:
243             raise NoAutenticado()
244         return usuario
