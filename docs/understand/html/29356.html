1     """Consume la API REST pública de TruchaSmart desplegada en Render (login JWT + endpoints)."""
2     
3     import json
4     import urllib.request
5     
6     URL = "https://truchasmart.onrender.com"
7     
8     
9     def pedir(ruta, datos=None, token=None):
10         cab = {"Content-Type": "application/json"}
11         if token:
12             cab["Authorization"] = f"Bearer {token}"
13         req = urllib.request.Request(
14             URL + ruta, data=json.dumps(datos).encode() if datos else None, headers=cab
15         )
16         with urllib.request.urlopen(req, timeout=120) as r:
17             return r.status, json.loads(r.read())
18     
19

23     )
24     tk = s["access_token"]
