20         def __init__(
21             self, estanques: RepositorioEstanque, lotes: RepositorioLote, reloj: Reloj

23             self._estanques = estanques
24             self._lotes = lotes

27         # -- estanques ---------------------------------------------------------
28         def crear(self, datos: DatosEstanque, solicitante: Usuario) -> Estanque:
29             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)
30             estanque = Estanque(
31                 nombre=datos.nombre,
32                 tipo=datos.tipo,
33                 volumen_m3=datos.volumen_m3,
34                 ubicacion=datos.ubicacion,
35                 creado_en=self._reloj.ahora(),
36             )
37             self._exigir_nombre_libre(estanque.nombre)
38             return self._estanques.guardar(estanque)
39     
40         def actualizar(self, estanque_id: int, datos: DatosEstanque, solicitante: Usuario) -> Estanque:
41             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)
42             actual = self._estanque(estanque_id)
43             nuevo = Estanque(
44                 nombre=datos.nombre,
45                 tipo=datos.tipo,
46                 volumen_m3=datos.volumen_m3,
47                 ubicacion=datos.ubicacion,
48                 creado_en=actual.creado_en,
49                 activo=actual.activo,
50                 id=actual.id,
51             )
52             if nuevo.nombre != actual.nombre:
53                 self._exigir_nombre_libre(nuevo.nombre)
54             return self._estanques.guardar(nuevo)
55     
56         def desactivar(self, estanque_id: int, solicitante: Usuario) -> None:
57             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)
58             estanque = self._estanque(estanque_id)
59             if self._lotes.activo_de(estanque_id) is not None:
60                 raise Conflicto("No se puede dar de baja un estanque con un lote activo")
61             estanque.activo = False
62             self._estanques.guardar(estanque)
63     
64         def obtener(self, estanque_id: int, solicitante: Usuario) -> Estanque:
65             solicitante.exigir(Permiso.CONSULTAR)
66             return self._estanque(estanque_id)
67     
68         def listar(self, solicitante: Usuario) -> list[Estanque]:

72         # -- lotes -------------------------------------------------------------
73         def sembrar_lote(self, comando: ComandoSembrarLote, solicitante: Usuario) -> Lote:
74             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)
75             estanque = self._estanque(comando.estanque_id)
76             lote = Lote.sembrar(
77                 estanque_id=estanque.id,
78                 codigo=comando.codigo,
79                 num_peces=comando.num_peces,
80                 peso_medio_g=comando.peso_medio_g,
81                 fecha_siembra=comando.fecha_siembra,
82                 hoy=self._reloj.ahora().date(),
83             )
84             if not estanque.activo:
85                 raise Conflicto(f"El estanque {estanque.nombre} está dado de baja")
86             existente = self._lotes.activo_de(estanque.id)
87             if existente is not None:
88                 raise Conflicto(
89                     f"El estanque {estanque.nombre} ya tiene el lote activo {existente.codigo}; "
90                     "ciérrelo antes de sembrar otro"
91                 )
92             if self._lotes.por_codigo(lote.codigo) is not None:
93                 raise Conflicto(f"Ya existe un lote con el código {lote.codigo}")
94             return self._lotes.guardar(lote)
95     
96         def lote_activo(self, estanque_id: int, solicitante: Usuario) -> Lote | None:
97             solicitante.exigir(Permiso.CONSULTAR)

99             return self._lotes.activo_de(estanque_id)

102             solicitante.exigir(Permiso.CONSULTAR)
103             self._estanque(estanque_id)
104             return self._lotes.de_estanque(estanque_id)
105     
106         def obtener_lote(self, lote_id: int, solicitante: Usuario) -> Lote:
107             solicitante.exigir(Permiso.CONSULTAR)
108             return self._lote(lote_id)
109     
110         def actualizar_peso(self, lote_id: int, peso_medio_g: float, solicitante: Usuario) -> Lote:
111             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)
112             lote = self._lote(lote_id)
113             lote.actualizar_peso(peso_medio_g)

115     
116         def cerrar_lote(self, lote_id: int, fecha: date, solicitante: Usuario) -> Lote:
117             solicitante.exigir(Permiso.GESTIONAR_ESTANQUES)

122         # -- auxiliares --------------------------------------------------------
123         def _estanque(self, estanque_id: int) -> Estanque:
124             estanque = self._estanques.por_id(estanque_id)
125             if estanque is None:
126                 raise NoEncontrado(f"No existe el estanque {estanque_id}")
127             return estanque
128     
129         def _lote(self, lote_id: int) -> Lote:
130             lote = self._lotes.por_id(lote_id)
131             if lote is None:
132                 raise NoEncontrado(f"No existe el lote {lote_id}")
133             return lote
134     
135         def _exigir_nombre_libre(self, nombre: str) -> None:
136             if self._estanques.por_nombre(nombre) is not None:
137                 raise Conflicto(f"Ya existe un estanque llamado '{nombre}'")
