24     class ServicioHistorial:
25         """Implementa el puerto de entrada ConsultarHistorial."""
26     
27         def __init__(
28             self,
29             lecturas: RepositorioLectura,
30             estanques: RepositorioEstanque,
31             alertas: RepositorioAlerta,
32             evaluador: EvaluadorCalidadAgua,
33             reloj: Reloj,
34         ) -> None:
35             self._lecturas = lecturas
36             self._estanques = estanques
37             self._alertas = alertas

41         def historial(
42             self,
43             estanque_id: int,
44             solicitante: Usuario,
45             desde: date | None = None,
46             hasta: date | None = None,
47         ) -> Historial:
48             solicitante.exigir(Permiso.CONSULTAR)
49             estanque = self._estanque(estanque_id)
50             hasta = hasta or self._reloj.ahora().date()
51             desde = desde or hasta - timedelta(days=DIAS_POR_DEFECTO - 1)
52             if desde > hasta:
53                 raise ErrorValidacion("desde", "La fecha 'desde' no puede ser posterior a 'hasta'")
54             if (hasta - desde).days + 1 > MAXIMO_DIAS:
55                 raise ErrorValidacion("desde", f"El rango máximo es de {MAXIMO_DIAS} días")
56             inicio = datetime.combine(desde, time.min, tzinfo=UTC)
57             fin = datetime.combine(hasta, time.max, tzinfo=UTC)
58             lecturas = sorted(
59                 self._lecturas.de_estanque(estanque_id, inicio, fin),
60                 key=lambda x: (x.medido_en, x.id or 0),
61             )
62             evaluadas = [LecturaEvaluada(x, self._evaluador.niveles(x)) for x in lecturas]
63             estadisticas = {}
64             for parametro in Parametro:
65                 valores = [x.valores()[parametro] for x in lecturas]
66                 if valores:
67                     estadisticas[parametro] = Estadistica(
68                         minimo=min(valores),
69                         maximo=max(valores),
70                         promedio=round(sum(valores) / len(valores), 2),
71                     )
72             return Historial(estanque, desde, hasta, evaluadas, estadisticas)

74         def riesgo(self, estanque_id: int, solicitante: Usuario) -> IndiceRiesgo:
75             solicitante.exigir(Permiso.CONSULTAR)
76             self._estanque(estanque_id)
77             ultima = self._lecturas.ultima_de(estanque_id)
78             niveles = self._evaluador.niveles(ultima) if ultima else {}
79             return riesgo_de(niveles, self._alertas.listar(estanque_id, EstadoAlerta.ABIERTA))
80     
81         def _estanque(self, estanque_id: int):
82             estanque = self._estanques.por_id(estanque_id)
83             if estanque is None:
84                 raise NoEncontrado(f"No existe el estanque {estanque_id}")
85             return estanque

88     def riesgo_de(niveles, alertas_abiertas) -> IndiceRiesgo:
89         """Índice de riesgo a partir de niveles y alertas ya consultados (evita repetir consultas)."""
90         criticas = sum(1 for a in alertas_abiertas if a.nivel is Nivel.CRITICA)
91         return calcular_indice_riesgo(niveles, criticas)
