1     """Crea el esquema (db/schema.sql) y carga los datos demo (db/seed.sql) en DATABASE_URL.
2     
3     - PostgreSQL: ejecuta los scripts tal cual (docker-compose ya lo hace al crear el volumen).
4     - SQLite: aplica una traducción mínima de dialecto (ADR-003):
5         SERIAL PRIMARY KEY -> INTEGER PRIMARY KEY AUTOINCREMENT, TIMESTAMPTZ -> TIMESTAMP,
6         se quita el sufijo "+00" de las fechas y se omite el bloque "-- postgresql-only".
7     
8     Uso:  python tools/init_db.py [--url sqlite:///./truchasmart.db] [--sin-seed] [--recrear]
9                                   [--si-vacia]
10     
11     --si-vacia: no hace nada si la tabla `usuarios` ya existe (la imagen Docker lo ejecuta al arrancar,
12     para que un PostgreSQL recién creado, como el de Render, quede con el esquema y los datos demo).
13     """
14     
15     from __future__ import annotations
16     
17     import argparse
18     import os
19     import re
20     import sqlite3
21     import sys
22     from pathlib import Path
23     
24     RAIZ = Path(__file__).resolve().parent.parent
25     sys.path.insert(0, str(RAIZ))
26     
27     from src.adapters.config import normalizar_url_bd  # noqa: E402
28     
29     SCHEMA = RAIZ / "db" / "schema.sql"
30     SEED = RAIZ / "db" / "seed.sql"
31     _MARCA_TIEMPO = re.compile(r"'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})\+00'")
32     
33     
34     def a_sqlite(sql: str) -> str:
35         sql = sql.split("-- postgresql-only")[0]
36         sql = sql.replace("SERIAL PRIMARY KEY", "INTEGER PRIMARY KEY AUTOINCREMENT")
37         sql = sql.replace("TIMESTAMPTZ", "TIMESTAMP")
38         # SQLAlchemy guarda las fechas en SQLite como 'AAAA-MM-DD HH:MM:SS.ffffff' y compara como
39         # texto: sin los microsegundos, '... 00:00:00' quedaría fuera de un rango que empieza en
40         # '... 00:00:00.000000' (Issue #19).
41         return _MARCA_TIEMPO.sub(lambda m: f"'{m.group(1)}.000000'", sql)
42     
43     
44     def cargar_sqlite(conexion: sqlite3.Connection, con_seed: bool = True) -> None:
45         conexion.execute("PRAGMA foreign_keys = ON")
46         conexion.executescript(a_sqlite(SCHEMA.read_text(encoding="utf-8")))
47         if con_seed:
48             conexion.executescript(a_sqlite(SEED.read_text(encoding="utf-8")))
49         conexion.commit()
50     
51     
52     def main(argv: list[str] | None = None) -> int:
53         parser = argparse.ArgumentParser(description="Inicializa la base de datos")
54         parser.add_argument("--url", default=os.getenv("DATABASE_URL", "sqlite:///./truchasmart.db"))
55         parser.add_argument("--sin-seed", action="store_true")
56         parser.add_argument("--recrear", action="store_true", help="SQLite: borra el archivo antes")
57         parser.add_argument("--si-vacia", action="store_true", help="solo si aún no hay esquema")
58         args = parser.parse_args(argv)
59         args.url = normalizar_url_bd(args.url)
60     
61         if args.url.startswith("sqlite:///"):
62             ruta = Path(args.url.removeprefix("sqlite:///"))
63             if args.recrear and ruta.exists():
64                 ruta.unlink()
65             with sqlite3.connect(ruta) as cx:
66                 existe = cx.execute(
67                     "SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = 'usuarios'"
68                 ).fetchone()
69                 if args.si_vacia and existe:
70                     print(f"SQLite ya inicializada en {ruta}; no se modifica.")
71                     return 0
72                 cargar_sqlite(cx, con_seed=not args.sin_seed)
73             print(f"SQLite inicializada en {ruta}")
74             return 0
75     
76         from sqlalchemy import create_engine, inspect
77     
78         motor = create_engine(args.url)
79         if args.si_vacia and inspect(motor).has_table("usuarios"):
80             print("Base de datos ya inicializada; no se modifica.")
81             return 0
82         with motor.begin() as cx:
83             cx.exec_driver_sql(SCHEMA.read_text(encoding="utf-8"))
84             if not args.sin_seed:
85                 cx.exec_driver_sql(SEED.read_text(encoding="utf-8"))
86         print("Base de datos inicializada.")
87         return 0
88     
89     
90     if __name__ == "__main__":
91         sys.exit(main())
