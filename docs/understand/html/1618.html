49         def crear(self, nombre: str, solicitante: Usuario, ip: str | None = None) -> ClaveCreada:
50             solicitante.exigir(Permiso.GESTIONAR_USUARIOS)
51             entidad, secreto = ClaveIntegracion.nueva(nombre, solicitante.id, self._reloj.ahora())
52             guardada = self._claves.guardar(entidad)
53             self._evento(
54                 TipoEvento.CLAVE_CREADA, solicitante, f"{guardada.nombre} ({guardada.prefijo}…)", ip
55             )
56             return ClaveCreada(guardada, secreto)

62         def revocar(
63             self, clave_id: int, solicitante: Usuario, ip: str | None = None
64         ) -> ClaveIntegracion:
65             solicitante.exigir(Permiso.GESTIONAR_USUARIOS)
66             clave = self._claves.por_id(clave_id)
67             if clave is None:
68                 raise NoEncontrado(f"No existe la clave de integración {clave_id}")
69             clave.revocar()
70             self._evento(
71                 TipoEvento.CLAVE_REVOCADA, solicitante, f"{clave.nombre} ({clave.prefijo}…)", ip
72             )
73             return self._claves.guardar(clave)

75         # -- consumo externo ---------------------------------------------------
76         def autenticar(self, secreto: str | None, ip: str | None = None) -> ClaveIntegracion:
77             """El rechazo se registra en la bitácora sin el valor de la clave (OWASP A09)."""
78             clave = self._claves.por_hash(huella(secreto)) if secreto else None
79             if clave is None or not clave.corresponde(secreto):
80                 if self._auditoria is not None:
81                     motivo = (
82                         "cabecera X-API-Key ausente"
83                         if not secreto
84                         else "clave no reconocida o revocada"
85                     )
86                     self._auditoria.registrar(TipoEvento.CLAVE_RECHAZADA, ip=ip, detalle=motivo)
87                 raise ClaveInvalida()
88             clave.ultimo_uso = self._reloj.ahora()
89             return self._claves.guardar(clave)

92             if self._auditoria is not None:

95         def estado_estanques(self, clave: ClaveIntegracion) -> list[EstadoExterno]:
96             self._exigir_activa(clave)
97             resultado = []
98             for e in self._estanques.listar(solo_activos=True):
99                 ultima = self._lecturas.ultima_de(e.id)
100                 abiertas = self._alertas.listar(e.id, EstadoAlerta.ABIERTA)
101                 resultado.append(
102                     EstadoExterno(
103                         estanque=e,
104                         ultima_lectura=ultima,
105                         niveles=self._evaluador.niveles(ultima) if ultima else {},
106                         alertas_abiertas=len(abiertas),
107                     )
108                 )
109             return resultado

111         def lecturas(
112             self, clave: ClaveIntegracion, estanque_id: int, desde: date | None, hasta: date | None
113         ) -> list[LecturaAgua]:
114             self._exigir_activa(clave)
115             if self._estanques.por_id(estanque_id) is None:
116                 raise NoEncontrado(f"No existe el estanque {estanque_id}")
117             hasta = hasta or self._reloj.ahora().date()
118             desde = desde or hasta - timedelta(days=DIAS_POR_DEFECTO - 1)
119             if desde > hasta:
120                 raise ErrorValidacion("desde", "La fecha 'desde' no puede ser posterior a 'hasta'")
121             if (hasta - desde).days + 1 > MAXIMO_DIAS:
122                 raise ErrorValidacion(
123                     "desde", f"El rango máximo de la API externa es {MAXIMO_DIAS} días"
124                 )
125             inicio = datetime.combine(desde, time.min, tzinfo=UTC)
126             fin = datetime.combine(hasta, time.max, tzinfo=UTC)
127             datos = self._lecturas.de_estanque(estanque_id, inicio, fin, MAXIMO_LECTURAS)
128             return sorted(datos, key=lambda x: (x.medido_en, x.id or 0))
129     
130         @staticmethod
131         def _exigir_activa(clave: ClaveIntegracion) -> None:
132             if not clave.activa:
133                 raise ClaveInvalida()
