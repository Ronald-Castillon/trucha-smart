23     @router.post("/{lote_id}/alimentacion", response_model=AlimentacionSalida, status_code=201)
24     def registrar_alimentacion(
25         lote_id: int,
26         datos: AlimentacionEntrada,
27         usuario: Usuario = Depends(usuario_actual),
28         c: Contenedor = Depends(contenedor),
29     ) -> AlimentacionSalida:
30         """Registra el alimento suministrado (kg > 0) a un lote activo."""
31         registro = c.produccion.registrar_alimentacion(
32             lote_id, datos.cantidad_kg, usuario, datos.registrado_en
33         )
34         c.confirmar()
35         return AlimentacionSalida.de(registro)
