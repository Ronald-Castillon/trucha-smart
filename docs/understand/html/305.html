58     def crear_simulador(app: FastAPI) -> SimuladorEnProceso:
59         """El paso del simulador abre su propia sesión y usa el caso de uso registrar_lectura."""
60     
61         def ejecutar_paso(estanque_id: int, usuario_id: int, m: Medicion) -> None:
62             recursos = app.state.recursos
63             sesion = recursos.bd.sesion()
64             try:
65                 c = Contenedor(sesion, recursos)
66                 usuario = c.usuarios.usuario_activo(usuario_id)
67                 if usuario is None:
68                     raise NoEncontrado("El usuario que inició la simulación ya no está activo")
69                 c.lecturas.registrar(
70                     ComandoRegistrarLectura(
71                         estanque_id=estanque_id,
72                         temperatura_c=m.temperatura_c,
73                         oxigeno_mg_l=m.oxigeno_mg_l,
74                         ph=m.ph,
75                         fuente=FuenteLectura.SENSOR,
76                     ),
77                     usuario,
78                 )
79                 c.confirmar()
80             except Exception:
81                 sesion.rollback()
82                 raise
83             finally:
84                 sesion.close()
85     
86         return SimuladorEnProceso(ejecutar_paso)

102     @router.post("", response_model=SimulacionSalida, status_code=201)
103     def iniciar_simulacion(
104         datos: SimulacionEntrada,
105         usuario: Usuario = Depends(usuario_actual),
106         c: Contenedor = Depends(contenedor),
107         sim: SimuladorEnProceso = Depends(simulador),
108     ) -> SimulacionSalida:
109         """Inicia (o reinicia) el sensor simulado de un estanque. Piscicultor o Administrador."""
110         usuario.exigir(Permiso.GESTIONAR_ESTANQUES)
111         estanque = c.estanques.obtener(datos.estanque_id, usuario)
112         if not estanque.activo:
113             raise Conflicto(f"El estanque {estanque.nombre} está dado de baja")
114         try:
115             estado = sim.iniciar(estanque.id, datos.escenario, datos.intervalo_s, usuario.id)
116         except ErrorSimulacion as e:
117             raise ErrorValidacion(
118                 "escenario" if "Escenario" in str(e) else "intervalo_s", str(e)
119             ) from e
120         return SimulacionSalida.de(estado)

123     @router.delete("/{estanque_id}", status_code=204)
124     def detener_simulacion(
125         estanque_id: int,
126         usuario: Usuario = Depends(usuario_actual),
127         sim: SimuladorEnProceso = Depends(simulador),
128     ) -> Response:
129         usuario.exigir(Permiso.GESTIONAR_ESTANQUES)
130         if not sim.detener(estanque_id):
131             raise NoEncontrado(f"No hay una simulación activa en el estanque {estanque_id}")
132         return Response(status_code=204)
