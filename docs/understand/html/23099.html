1     """Verifica la pureza del dominio hexagonal (métrica M1.1).
2     
3     Recorre con AST cada módulo .py de src/domain (y, como control adicional, src/ports) y
4     marca como "contaminado" todo módulo que importe algo que no sea:
5       - la biblioteca estándar de Python (sys.stdlib_module_names), o
6       - src.domain / src.ports (incluidas importaciones relativas que se resuelvan dentro de ellos).
7     
8     Imprime  M1.1 = módulos contaminados / total de módulos de src/domain x 100
9     y termina con código 1 si M1.1 > 0 o si algún puerto está contaminado.
10     
11     Uso:  python tools/check_domain_purity.py [--raiz .]
12     """
13     
14     from __future__ import annotations
15     
16     import argparse
17     import ast
18     import sys
19     from dataclasses import dataclass, field
20     from pathlib import Path
21     
22     PERMITIDOS_INTERNOS = ("src.domain", "src.ports")
23     
24     
25     @dataclass
26     class ResultadoModulo:
27         ruta: Path
28         imports_prohibidos: list[str] = field(default_factory=list)
29     
30         @property
31         def contaminado(self) -> bool:
32             return bool(self.imports_prohibidos)
33     
34     
35     @dataclass
36     class ResultadoPaquete:
37         paquete: str
38         modulos: list[ResultadoModulo]
39     
40         @property
41         def total(self) -> int:
42             return len(self.modulos)
43     
44         @property
45         def contaminados(self) -> list[ResultadoModulo]:
46             return [m for m in self.modulos if m.contaminado]
47     
48         @property
49         def porcentaje(self) -> float:
50             return 0.0 if self.total == 0 else len(self.contaminados) / self.total * 100
51     
52     
53     def _nombre_modulo(raiz: Path, ruta: Path) -> str:
54         partes = list(ruta.relative_to(raiz).with_suffix("").parts)
55         if partes[-1] == "__init__":
56             partes.pop()
57         return ".".join(partes)
58     
59     
60     def _resolver_relativo(modulo_actual: str, es_paquete: bool, nivel: int, nombre: str | None) -> str:
61         base = modulo_actual.split(".")
62         if not es_paquete:
63             base = base[:-1]
64         if nivel > 1:
65             base = base[: len(base) - (nivel - 1)]
66         return ".".join(base + ([nombre] if nombre else []))
67     
68     
69     def _es_permitido(nombre: str) -> bool:
70         if any(nombre == p or nombre.startswith(p + ".") for p in PERMITIDOS_INTERNOS):
71             return True
72         return nombre.split(".")[0] in sys.stdlib_module_names
73     
74     
75     def analizar_modulo(raiz: Path, ruta: Path) -> ResultadoModulo:
76         modulo = _nombre_modulo(raiz, ruta)
77         es_paquete = ruta.name == "__init__.py"
78         arbol = ast.parse(ruta.read_text(encoding="utf-8"), filename=str(ruta))
79         resultado = ResultadoModulo(ruta=ruta)
80         for nodo in ast.walk(arbol):
81             nombres: list[str] = []
82             if isinstance(nodo, ast.Import):
83                 nombres = [a.name for a in nodo.names]
84             elif isinstance(nodo, ast.ImportFrom):
85                 if nodo.level:
86                     nombres = [_resolver_relativo(modulo, es_paquete, nodo.level, nodo.module)]
87                 elif nodo.module:
88                     nombres = [nodo.module]
89             for nombre in nombres:
90                 if not _es_permitido(nombre):
91                     resultado.imports_prohibidos.append(f"línea {nodo.lineno}: {nombre}")
92         return resultado
93     
94     
95     def analizar_paquete(raiz: Path, paquete: str) -> ResultadoPaquete:
96         carpeta = raiz / Path(*paquete.split("."))
97         modulos = [analizar_modulo(raiz, p) for p in sorted(carpeta.rglob("*.py"))]
98         return ResultadoPaquete(paquete=paquete, modulos=modulos)
99     
100     
101     def main(argv: list[str] | None = None) -> int:
102         parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
103         parser.add_argument("--raiz", default=Path(__file__).resolve().parent.parent, type=Path)
104         args = parser.parse_args(argv)
105         raiz: Path = args.raiz.resolve()
106         if hasattr(sys.stdout, "reconfigure"):
107             sys.stdout.reconfigure(encoding="utf-8")  # consola de Windows
108     
109         dominio = analizar_paquete(raiz, "src.domain")
110         puertos = analizar_paquete(raiz, "src.ports")
111     
112         for res in (dominio, puertos):
113             print(f"{res.paquete}: {res.total} módulos, {len(res.contaminados)} contaminados")
114             for m in res.contaminados:
115                 print(f"  ✗ {m.ruta.relative_to(raiz)}")
116                 for imp in m.imports_prohibidos:
117                     print(f"      {imp}")
118     
119         print(
120             f"M1.1 = {len(dominio.contaminados)} / {dominio.total} x 100 = {dominio.porcentaje:.1f} %"
121         )
122         if dominio.contaminados or puertos.contaminados:
123             print("FALLO: el dominio o los puertos importan dependencias no permitidas.")
124             return 1
125         print("OK: dominio y puertos usan solo la biblioteca estándar.")
126         return 0
127     
128     
129     if __name__ == "__main__":
130         sys.exit(main())
