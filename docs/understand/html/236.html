48     def crear_recursos(config: Configuracion) -> Recursos:
49         config.validar()  # RNF-+: no arrancar en producción con la clave JWT de ejemplo
50         bd = BaseDatos(config.database_url, salud_timeout_s=config.salud_timeout_s)
51         bd.crear_tablas()
52         return Recursos(
53             bd=bd,
54             hasher=HasherBcrypt(),
55             tokens=ServicioTokens(config.jwt_secret, config.jwt_expira_minutos),
56             reloj=RelojSistema(),
57             umbrales=cargar_umbrales(config.umbrales_path),
58             notificador=NotificadorLog(),
59             alimentacion=cargar_alimentacion(config.alimentacion_path),
60             generador_reportes=GeneradorReportesArchivo(),
61             cookie_segura=config.es_produccion,
62             modelo_riesgo=ModeloLogistico.desde_archivo(config.modelo_riesgo_path),
63         )

66     def crear_app(config: Configuracion | None = None, recursos: Recursos | None = None) -> FastAPI:
67         """Crea la app. Los recursos (BD, etc.) se construyen al arrancar si no se inyectan."""
68     
69         @asynccontextmanager
70         async def ciclo_de_vida(app: FastAPI) -> AsyncIterator[None]:
71             if app.state.recursos is None:
72                 app.state.recursos = crear_recursos(config or Configuracion.desde_entorno())
73             yield
74             app.state.simulador.detener_todas()
75     
76         app = FastAPI(
77             title="TruchaSmart API",
78             version="1.0.0",
79             description=DESCRIPCION,
80             lifespan=ciclo_de_vida,
81         )
82         app.state.recursos = recursos
83         app.state.simulador = simulador.crear_simulador(app)
84         registrar_manejadores(app)
85         app.add_middleware(CabecerasSeguridad)
86         app.include_router(salud.router)
87         app.include_router(simulador.router)
88         app.include_router(auth.router)
89         app.include_router(estanques.router)
90         app.include_router(lecturas.router)
91         app.include_router(alertas.router)
92         app.include_router(historial.router)
93         app.include_router(prediccion.router)
94         app.include_router(riesgo_modelo.router)
95         app.include_router(produccion.router)
96         app.include_router(racion.router)
97         app.include_router(reportes.router)
98         app.include_router(integraciones.admin)
99         app.include_router(integraciones.externa)
100         app.include_router(web.router)
101         app.mount("/static", StaticFiles(directory=CARPETA_WEB / "static"), name="static")
102         return app
103
