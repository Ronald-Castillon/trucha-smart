19     
20         def __init__(
21             self,
22             lotes: RepositorioLote,
23             mortalidad: RepositorioMortalidad,
24             lecturas: RepositorioLectura,
25             evaluador: EvaluadorCalidadAgua,
26             recomendador: RecomendadorRacion,
27         ) -> None:
28             self._lotes = lotes
29             self._mortalidad = mortalidad
30             self._lecturas = lecturas
31             self._evaluador = evaluador

33     
34         def recomendar(self, lote_id: int, solicitante: Usuario) -> RecomendacionLote:
35             solicitante.exigir(Permiso.CONSULTAR)
36             lote = self._lotes.por_id(lote_id)
37             if lote is None:
38                 raise NoEncontrado(f"No existe el lote {lote_id}")
39             if not lote.activo:
40                 raise Conflicto(f"El lote {lote.codigo} está cerrado")
41             vivos = peces_vivos(lote, self._mortalidad.total(lote_id))
42             lectura = self._lecturas.ultima_de(lote.estanque_id)
43             niveles = self._evaluador.niveles(lectura) if lectura else {}
44             recomendacion = self._recomendador.recomendar(
45                 biomasa_kg=biomasa_kg(vivos, lote.peso_medio_actual_g),
46                 peso_medio_g=lote.peso_medio_actual_g,
47                 nivel_temperatura=niveles.get(Parametro.TEMPERATURA),
48                 nivel_oxigeno=niveles.get(Parametro.OXIGENO_DISUELTO),
49             )
50             return RecomendacionLote(lote, vivos, lectura, recomendacion)
