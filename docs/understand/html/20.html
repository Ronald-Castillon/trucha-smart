45     LONGITUD_MINIMA_PASSWORD = 8
46     MAXIMO_INTENTOS_FALLIDOS = 3  # RF-11 / OWASP A07
47     MINUTOS_BLOQUEO = 5
48

50     @dataclass
51     class Usuario:
52         nombre_usuario: str
53         nombre_completo: str
54         password_hash: str
55         rol: Rol
56         creado_en: datetime
57         activo: bool = True
58         id: int | None = None
59         intentos_fallidos: int = 0
60         bloqueado_hasta: datetime | None = None
61     
62         def puede(self, permiso: Permiso) -> bool:
63             return self.activo and permiso in PERMISOS_POR_ROL[self.rol]
64     
65         def exigir(self, permiso: Permiso) -> None:
66             if not self.puede(permiso):
67                 raise PermisoDenegado(
68                     f"El rol {self.rol.value} no tiene permiso para {permiso.value.lower()}"
69                 )
70     
71         # -- protección contra fuerza bruta (RF-11) ------------------------------
72         def bloqueado(self, ahora: datetime) -> bool:
73             return self.bloqueado_hasta is not None and ahora < self.bloqueado_hasta
74     
75         def registrar_fallo(self, ahora: datetime) -> bool:
76             """Suma un intento fallido; al llegar al máximo bloquea la cuenta. True si se bloqueó."""
77             self.intentos_fallidos += 1
78             if self.intentos_fallidos >= MAXIMO_INTENTOS_FALLIDOS:
79                 self.intentos_fallidos = 0
80                 self.bloqueado_hasta = ahora + timedelta(minutes=MINUTOS_BLOQUEO)
81                 return True
82             return False
83     
84         def registrar_exito(self) -> None:
85             self.intentos_fallidos = 0
86             self.bloqueado_hasta = None
87

89     def validar_nombre_usuario(nombre: str) -> str:
90         limpio = nombre.strip().lower()
91         if not (3 <= len(limpio) <= 50) or not all(c.isalnum() or c in "._-" for c in limpio):
92             raise ErrorValidacion(
93                 "nombre_usuario",
94                 "El nombre de usuario debe tener 3 a 50 caracteres (letras, números, . _ -)",
95             )
96         return limpio

99     def validar_password(password: str) -> None:
100         if len(password) < LONGITUD_MINIMA_PASSWORD:
101             raise ErrorValidacion(
102                 "password",
103                 f"La contraseña debe tener al menos {LONGITUD_MINIMA_PASSWORD} caracteres",
104             )
