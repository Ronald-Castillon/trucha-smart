61     def _cumple(valor: float | None, inc: int, metrica: str) -> bool | None:
62         if valor is None:
63             return None
64         op, meta = g.METAS[metrica][inc]
65         return valor <= meta if op == "<=" else valor >= meta

71     def calc_m1_1(raiz: Path, inc: int) -> tuple[str, str, float]:
72         dom = analizar_paquete(raiz, "src.domain")
73         num, den = len(dom.contaminados), dom.total
74         return f"{num} módulos contaminados", f"{den} módulos", num / den * 100

77     def calc_m1_2(raiz: Path, inc: int) -> tuple[str, str, float]:
78         exe = shutil.which("lint-imports")
79         r = subprocess.run([exe], cwd=raiz, capture_output=True, text=True, encoding="utf-8",
80                            env={**os.environ, "PYTHONIOENCODING": "utf-8"})  # fmt: skip
81         contratos_ok = r.returncode == 0
82         contaminados = {
83             m.ruta.resolve()
84             for paq in ("src.domain", "src.ports")
85             for m in analizar_paquete(raiz, paq).contaminados
86         }
87         conformes = 0
88         for carpetas in g.COMPONENTES.values():
89             modulos = [p for c in carpetas for p in (raiz / c).rglob("*.py") if p.name != "__init__.py"]
90             presentes = all(any((raiz / c).rglob("*.py")) for c in carpetas) and bool(modulos)
91             puros = not any(p.resolve() in contaminados for p in modulos)
92             conformes += presentes and puros and contratos_ok
93         estado = "KEPT" if contratos_ok else "BROKEN"
94         return (
95             f"{conformes} componentes conformes (contratos {estado})",
96             "5 componentes",
97             conformes / 5 * 100,
98         )

101     def calc_m2_1(raiz: Path, inc: int) -> tuple[str, str, float]:
102         esperadas = [
103             raiz / "evidencias" / f"inc{inc}" / f"RF{rf:02d}_paso{n}_{s}.png"
104             for rf in g.RF_POR_INCREMENTO[inc]
105             for n, s in ((1, "entrada"), (2, "resultado"))
106         ]
107         existentes = sum(p.is_file() and p.stat().st_size > 0 for p in esperadas)
108         return (
109             f"{existentes} capturas existentes",
110             f"{len(esperadas)} esperadas",
111             existentes / len(esperadas) * 100,
112         )

115     def calc_m2_2(raiz: Path, inc: int) -> tuple[str, str, float]:
116         r = subprocess.run(
117             ["gh", "issue", "list", "--state", "all", "--label", "bug", "--label", f"inc{inc}",
118              "--json", "number"], cwd=raiz, capture_output=True, text=True, encoding="utf-8",
119         )  # fmt: skip
120         numeros = [i["number"] for i in json.loads(r.stdout or "[]")]
121         rf = len(g.RF_POR_INCREMENTO[inc])
122         ids = ", ".join(f"#{n}" for n in numeros) or "ninguno"
123         return f"{len(numeros)} defecto(s): {ids}", f"{rf} RF", len(numeros) / rf

126     def calc_m2_3(raiz: Path, inc: int) -> tuple[str, str, float]:
127         raiz_xml = ET.parse(raiz / "coverage.xml").getroot()
128         cubiertas = total = 0
129         for clase in raiz_xml.iter("class"):
130             for linea in clase.iter("line"):
131                 total += 1
132                 cubiertas += int(linea.get("hits", "0")) > 0
133         return f"{cubiertas} líneas ejecutadas", f"{total} líneas ejecutables", cubiertas / total * 100

136     def calc_m3_1(raiz: Path, inc: int) -> tuple[str, str, float]:
137         datos = yaml.safe_load((raiz / "docs" / "gqm" / "brechas.yaml").read_text(encoding="utf-8"))
138         pruebas = "\n".join(p.read_text(encoding="utf-8") for p in (raiz / "tests").rglob("*.py"))
139         resueltas = []
140         for b in datos["brechas"]:
141             if b.get("resuelta_en") is None or b["resuelta_en"] > inc:
142                 continue
143             ok = True
144             for rf in b["rf"]:
145                 n = int(rf.split("-")[1])
146                 carpeta = raiz / "evidencias" / f"inc{1 if n <= 4 else 2}"
147                 ok &= all((carpeta / f"RF{n:02d}_paso{k}_{s}.png").is_file()
148                           for k, s in ((1, "entrada"), (2, "resultado")))  # fmt: skip
149                 ok &= f"CP-RF{n:02d}-01" in pruebas.upper().replace("_", "-") and (
150                     f"CP-RF{n:02d}-02" in pruebas.upper().replace("_", "-")
151                 )
152             if ok:
153                 resueltas.append(b["id"])
154         total = len(datos["brechas"])
155         return (
156             f"{len(resueltas)} resueltas ({', '.join(resueltas)})",
157             f"{total} brechas",
158             len(resueltas) / total * 100,
159         )

162     def calc_m3_2(raiz: Path, inc: int) -> tuple[str, str, float | None]:
163         # Las mediciones de valor se registraron en main (PR #22): se usan para ambos incrementos.
164         datos = yaml.safe_load(
165             (RAIZ / "docs" / "gqm" / "valor_indicadores.yaml").read_text(encoding="utf-8")
166         )
167         mejoras, partes = [], []
168         for iv in datos["indicadores"]:
169             a, t = iv.get("as_is"), iv.get(f"to_be_inc{inc}")
170             if a in (None, "") or t in (None, ""):
171                 partes.append(f"{iv['id']}: sin dato")
172                 continue
173             a, t = float(a), float(t)
174             m = (
175                 (a - t) / a * 100
176                 if iv["sentido"] == "menor_es_mejor"
177                 else min(100.0, (t - a) / a * 100)
178             )
179             mejoras.append(m)
180             partes.append(f"{iv['id']} {m:.1f} % ({iv.get('as_is_tipo')})")
181         valor = sum(mejoras) / len(mejoras) if mejoras else None
182         return "; ".join(partes), f"{len(mejoras)} de {len(datos['indicadores'])} IV con dato", valor

196     def valor_herramienta(metrica: str, raiz: Path, inc: int) -> tuple[float | None, str | None]:
197         g.RAIZ = raiz if metrica != "M3.2" else RAIZ
198         funcion = CALCULOS[metrica][1]
199         m = funcion(inc) if metrica in ("M2.1", "M2.2", "M3.1", "M3.2") else funcion()
200         return m.valor, m.advertencia

203     def verificar(metrica: str, raices: dict[int, Path]) -> bool:
204         nombre, formula, fuente, meta1, meta2 = MATRIZ[metrica]
205         calcular, _, formato = CALCULOS[metrica]
206         print(f"{metrica} · {nombre}")
207         print(f"  Fórmula (Matriz 3.1): {formula}")
208         print(f"  Fuente de datos:      {fuente}")
209         todo_ok = True
210         for inc, meta_matriz in ((1, meta1), (2, meta2)):
211             raiz = raices[inc]
212             num, den, valor = calcular(raiz, inc)
213             herramienta, advertencia = valor_herramienta(metrica, raiz, inc)
214             op, objetivo = g.METAS[metrica][inc]
215             coincide = (valor is None and herramienta is None) or (
216                 valor is not None and herramienta is not None and abs(valor - herramienta) < 1e-6
217             )
218             cumple = _cumple(valor, inc, metrica)
219             if advertencia:
220                 estado = f"NO ACREDITA ({advertencia})"
221             else:
222                 estado = {True: "CUMPLE", False: "NO CUMPLE", None: "PENDIENTE"}[cumple]
223             texto = "sin dato" if valor is None else formato.format(valor)
224             print(f"  Incremento {inc} ({'v1.0-PMV' if inc == 1 else 'main'})")
225             print(f"    Numerador:    {num}")
226             print(f"    Denominador:  {den}")
227             print(f"    Resultado:    {texto}")
228             print(
229                 f"    Meta Matriz 3.1: {meta_matriz}  ·  meta en gqm_metrics.py: {op} {formato.format(objetivo)}"
230             )
231             print(f"    gqm_metrics.py reporta: {'sin dato' if herramienta is None else formato.format(herramienta)}"
232                   f"  ->  {'COINCIDE' if coincide else 'NO COINCIDE'}")  # fmt: skip
233             print(f"    Estado: {estado}")
234             todo_ok &= coincide
235         print()
236         return todo_ok

239     def main(argv: list[str] | None = None) -> int:
240         parser = argparse.ArgumentParser(description="Verifica las fórmulas GQM de la Matriz 3.1")
241         parser.add_argument("--inc1", type=Path, required=True, help="checkout del tag v1.0-PMV")
242         parser.add_argument("--inc2", type=Path, default=RAIZ)
243         parser.add_argument("--metrica", choices=list(MATRIZ), help="solo una métrica")
244         args = parser.parse_args(argv)
245         if hasattr(sys.stdout, "reconfigure"):
246             sys.stdout.reconfigure(encoding="utf-8")
247         raices = {1: args.inc1.resolve(), 2: args.inc2.resolve()}
248         metricas = [args.metrica] if args.metrica else list(MATRIZ)
249         ok = all([verificar(m, raices) for m in metricas])
250         print("Todas las fórmulas coinciden con gqm_metrics.py." if ok else "HAY DIFERENCIAS.")
251         return 0 if ok else 1
