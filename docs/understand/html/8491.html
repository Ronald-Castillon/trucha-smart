43     def _estadistica(valores: list[float]) -> tuple[float, float, float] | None:
44         if not valores:
45             return None
46         return (min(valores), round(sum(valores) / len(valores), 2), max(valores))

92         def armar(self, tipo: TipoReporte, desde: date, hasta: date, solicitante: Usuario) -> Reporte:
93             solicitante.exigir(Permiso.GESTIONAR_REPORTES)
94             if desde > hasta:
95                 raise ErrorValidacion("desde", "La fecha 'desde' no puede ser posterior a 'hasta'")
96             if (hasta - desde).days + 1 > MAXIMO_DIAS:
97                 raise ErrorValidacion("desde", f"El rango máximo es de {MAXIMO_DIAS} días")
98             inicio = datetime.combine(desde, time.min, tzinfo=UTC)
99             fin = datetime.combine(hasta, time.max, tzinfo=UTC)
100             filas_calidad, resumen_calidad, filas_produccion = [], [], []
101             if tipo is TipoReporte.CALIDAD_AGUA:
102                 filas_calidad, resumen_calidad = self._calidad(inicio, fin)
103             else:
104                 filas_produccion = self._produccion(desde, hasta, inicio, fin)
105             return Reporte(
106                 tipo=tipo,
107                 titulo=TITULOS[tipo],
108                 desde=desde,
109                 hasta=hasta,
110                 generado_en=self._reloj.ahora(),
111                 generado_por=solicitante.nombre_completo,
112                 filas_calidad=filas_calidad,
113                 resumen_calidad=resumen_calidad,
114                 filas_produccion=filas_produccion,
115             )

117         def _calidad(self, inicio: datetime, fin: datetime):
118             filas, resumen = [], []
119             for e in self._estanques.listar(solo_activos=False):
120                 lecturas = sorted(
121                     self._lecturas.de_estanque(e.id, inicio, fin), key=lambda x: (x.medido_en, x.id)
122                 )
123                 for x in lecturas:
124                     filas.append(
125                         FilaCalidad(
126                             estanque=e.nombre,
127                             medido_en=x.medido_en,
128                             temperatura_c=x.temperatura_c,
129                             oxigeno_mg_l=x.oxigeno_mg_l,
130                             ph=x.ph,
131                             fuente=x.fuente.value,
132                             niveles=self._evaluador.niveles(x),
133                         )
134                     )
135                 alertas = [a for a in self._alertas.listar(e.id) if inicio <= a.creada_en <= fin]
136                 if not lecturas and not alertas:
137                     continue
138                 resumen.append(
139                     ResumenCalidadEstanque(
140                         estanque=e.nombre,
141                         lecturas=len(lecturas),
142                         temperatura=_estadistica([x.temperatura_c for x in lecturas]),
143                         oxigeno=_estadistica([x.oxigeno_mg_l for x in lecturas]),
144                         ph=_estadistica([x.valores()[Parametro.PH] for x in lecturas]),
145                         alertas_precaucion=sum(a.nivel is Nivel.PRECAUCION for a in alertas),
146                         alertas_criticas=sum(a.nivel is Nivel.CRITICA for a in alertas),
147                         alertas_atendidas=sum(a.estado is EstadoAlerta.ATENDIDA for a in alertas),
148                     )
149                 )
150             return filas, resumen

152         def _produccion(self, desde: date, hasta: date, inicio: datetime, fin: datetime):
153             filas = []
154             for e in self._estanques.listar(solo_activos=False):
155                 for lote in sorted(self._lotes.de_estanque(e.id), key=lambda lo: lo.fecha_siembra):
156                     vigente = lote.fecha_siembra <= hasta and (
157                         lote.fecha_cierre is None or lote.fecha_cierre >= desde
158                     )
159                     if not vigente:
160                         continue
161                     mortalidad_total = self._mortalidad.total(lote.id)
162                     r = resumen_produccion(lote, mortalidad_total, self._alimentacion.total_kg(lote.id))
163                     filas.append(
164                         FilaProduccion(
165                             estanque=e.nombre,
166                             lote=lote.codigo,
167                             fecha_siembra=lote.fecha_siembra,
168                             activo=lote.activo,
169                             peces_sembrados=lote.num_peces_inicial,
170                             mortalidad_periodo=sum(
171                                 m.cantidad for m in self._mortalidad.de_lote(lote.id, inicio, fin)
172                             ),
173                             mortalidad_total=mortalidad_total,
174                             peces_vivos=r.peces_vivos,
175                             peso_medio_actual_g=lote.peso_medio_actual_g,
176                             biomasa_kg=r.biomasa_kg,
177                             alimento_periodo_kg=round(
178                                 sum(
179                                     a.cantidad_kg
180                                     for a in self._alimentacion.de_lote(lote.id, inicio, fin)
181                                 ),
182                                 3,
183                             ),
184                             alimento_total_kg=r.alimento_total_kg,
185                             fcr=r.fcr_texto,
186                         )
187                     )
188             return filas
