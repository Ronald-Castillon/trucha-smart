1     """Índice de riesgo de un estanque (RF-05, brecha B-07).
2     
3     riesgo = min(100, 40 x parámetros CRÍTICOS en la última lectura
4                       + 15 x parámetros en PRECAUCIÓN en la última lectura
5                       + 10 x alertas CRÍTICAS abiertas)
6     Semáforo: 0-29 verde (bajo), 30-59 amarillo (medio), >= 60 rojo (alto).
7     """
8     
9     from __future__ import annotations
10     
11     from dataclasses import dataclass
12     from enum import StrEnum
13     
14     from src.domain.model.lectura_agua import Parametro
15     from src.domain.model.umbral import Nivel
16     
17     PESO_CRITICO = 40
18     PESO_PRECAUCION = 15
19     PESO_ALERTA_CRITICA = 10
20     MAXIMO = 100
21     LIMITE_MEDIO = 30
22     LIMITE_ALTO = 60
23     
24     
25     class Semaforo(StrEnum):
26         VERDE = "VERDE"
27         AMARILLO = "AMARILLO"
28         ROJO = "ROJO"
29     
30     
31     ETIQUETAS = {Semaforo.VERDE: "bajo", Semaforo.AMARILLO: "medio", Semaforo.ROJO: "alto"}
32     
33     
34     @dataclass(frozen=True)
35     class IndiceRiesgo:
36         valor: int
37         semaforo: Semaforo
38         parametros_criticos: int
39         parametros_precaucion: int
40         alertas_criticas_abiertas: int
41     
42         @property
43         def nivel(self) -> str:
44             return ETIQUETAS[self.semaforo]
45     
46         @property
47         def formula(self) -> str:
48             bruto = (
49                 PESO_CRITICO * self.parametros_criticos
50                 + PESO_PRECAUCION * self.parametros_precaucion
51                 + PESO_ALERTA_CRITICA * self.alertas_criticas_abiertas
52             )
53             return (
54                 f"min(100, {PESO_CRITICO}×{self.parametros_criticos} + "
55                 f"{PESO_PRECAUCION}×{self.parametros_precaucion} + "
56                 f"{PESO_ALERTA_CRITICA}×{self.alertas_criticas_abiertas}) = min(100, {bruto}) "
57                 f"= {self.valor}"
58             )
59     
60     
61     def semaforo_de(valor: int) -> Semaforo:
62         if valor >= LIMITE_ALTO:
63             return Semaforo.ROJO
64         if valor >= LIMITE_MEDIO:
65             return Semaforo.AMARILLO
66         return Semaforo.VERDE
67     
68     
69     def calcular_indice_riesgo(
70         niveles_ultima_lectura: dict[Parametro, Nivel], alertas_criticas_abiertas: int
71     ) -> IndiceRiesgo:
72         if alertas_criticas_abiertas < 0:
73             raise ValueError("El número de alertas no puede ser negativo")
74         criticos = sum(1 for n in niveles_ultima_lectura.values() if n is Nivel.CRITICA)
75         precaucion = sum(1 for n in niveles_ultima_lectura.values() if n is Nivel.PRECAUCION)
76         valor = min(
77             MAXIMO,
78             PESO_CRITICO * criticos
79             + PESO_PRECAUCION * precaucion
80             + PESO_ALERTA_CRITICA * alertas_criticas_abiertas,
81         )
82         return IndiceRiesgo(valor, semaforo_de(valor), criticos, precaucion, alertas_criticas_abiertas)
