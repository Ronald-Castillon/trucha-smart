30         def guardar(self, usuario: Usuario) -> Usuario:
31             fila = self._s.get(UsuarioORM, usuario.id) if usuario.id else None

40             fila.creado_en = usuario.creado_en
41             fila.intentos_fallidos = usuario.intentos_fallidos
42             fila.bloqueado_hasta = usuario.bloqueado_hasta
43             self._s.flush()
