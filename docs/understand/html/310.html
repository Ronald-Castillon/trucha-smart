1     """Cabeceras de seguridad HTTP (RNF-+, OWASP A02 Security Misconfiguration).
2     
3     La CSP permite scripts propios, Chart.js desde cdn.jsdelivr.net y los scripts en línea de las
4     plantillas ('unsafe-inline': deuda registrada en docs/calidad/seguridad_owasp_2025.md).
5     Swagger UI (/docs) y ReDoc (/redoc) cargan recursos de CDN y quedan sin CSP.
6     """
7     
8     from __future__ import annotations
9     
10     from starlette.middleware.base import BaseHTTPMiddleware
11     from starlette.requests import Request
12     
13     CSP = "; ".join(
14         [
15             "default-src 'self'",
16             "script-src 'self' 'unsafe-inline' https://cdn.jsdelivr.net",
17             "style-src 'self' 'unsafe-inline'",
18             "img-src 'self' data:",
19             "connect-src 'self'",
20             "frame-ancestors 'none'",
21             "base-uri 'self'",
22             "form-action 'self'",
23             "object-src 'none'",
24         ]
25     )
26     CABECERAS = {
27         "X-Content-Type-Options": "nosniff",
28         "X-Frame-Options": "DENY",
29         "Referrer-Policy": "strict-origin-when-cross-origin",
30         "Permissions-Policy": "camera=(), microphone=(), geolocation=()",
31         "Cross-Origin-Opener-Policy": "same-origin",
32     }
33     SIN_CSP = ("/docs", "/redoc")
34     
35     
36     class CabecerasSeguridad(BaseHTTPMiddleware):
37         async def dispatch(self, request: Request, call_next):
38             respuesta = await call_next(request)
39             for nombre, valor in CABECERAS.items():
40                 respuesta.headers.setdefault(nombre, valor)
41             if not request.url.path.startswith(SIN_CSP):
42                 respuesta.headers.setdefault("Content-Security-Policy", CSP)
43             if request.url.path.startswith("/api/"):
44                 respuesta.headers.setdefault("Cache-Control", "no-store")
45             return respuesta
