5     import math
6     import random
7     from collections.abc import Iterator
8     from dataclasses import dataclass
9     
10     ESCENARIOS = ("normal", "critico_oxigeno", "critico_temperatura")
11     
12     
13     @dataclass(frozen=True)
14     class Medicion:
15         temperatura_c: float
16         oxigeno_mg_l: float
17         ph: float
18     
19     
20     def _r(x: float) -> float:
21         return round(x, 2)
22     
23     
24     def generar(escenario: str, semilla: int | None = None) -> Iterator[Medicion]:
25         """Genera mediciones sucesivas del escenario indicado.
26     
27         - normal: valores dentro de los rangos normales con pequeña variación.
28         - critico_oxigeno: el OD cae 0.4 mg/L por paso desde 8.0 hasta 3.5 (pasa por
29           precaución y llega a crítico) y se mantiene allí.
30         - critico_temperatura: la temperatura sube 0.5 °C por paso desde 14.0 hasta 20.0.
31         """
32         if escenario not in ESCENARIOS:
33             raise ValueError(f"Escenario desconocido: {escenario}. Opciones: {', '.join(ESCENARIOS)}")
34         rnd = random.Random(semilla)
35         paso = 0
36         while True:
37             onda = math.sin(paso / 6)
38             temperatura = 12.5 + 1.2 * onda + rnd.uniform(-0.2, 0.2)
39             oxigeno = 8.3 - 0.4 * onda + rnd.uniform(-0.2, 0.2)
40             ph = 7.3 + rnd.uniform(-0.2, 0.2)
41             if escenario == "critico_oxigeno":
42                 oxigeno = max(3.5, 8.0 - 0.4 * paso)
43             elif escenario == "critico_temperatura":
44                 temperatura = min(20.0, 14.0 + 0.5 * paso)
45             yield Medicion(_r(temperatura), _r(oxigeno), _r(ph))
46             paso += 1
