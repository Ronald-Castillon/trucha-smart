1     """Exporta el contrato OpenAPI de la aplicación a docs/openapi.json.
2     
3     Uso:  python tools/export_openapi.py          # regenera el archivo
4           python tools/export_openapi.py --check  # falla (exit 1) si el archivo está desactualizado
5     """
6     
7     from __future__ import annotations
8     
9     import argparse
10     import json
11     import sys
12     from pathlib import Path
13     
14     RAIZ = Path(__file__).resolve().parent.parent
15     DESTINO = RAIZ / "docs" / "openapi.json"
16     sys.path.insert(0, str(RAIZ))
17     
18     
19     def contrato_actual() -> str:
20         from src.main import crear_app
21     
22         return json.dumps(crear_app().openapi(), indent=2, ensure_ascii=False, sort_keys=True) + "\n"
23     
24     
25     def main(argv: list[str] | None = None) -> int:
26         parser = argparse.ArgumentParser(description="Exporta docs/openapi.json")
27         parser.add_argument("--check", action="store_true", help="solo verificar, no escribir")
28         args = parser.parse_args(argv)
29         actual = contrato_actual()
30         if args.check:
31             guardado = DESTINO.read_text(encoding="utf-8") if DESTINO.exists() else ""
32             if guardado != actual:
33                 print("docs/openapi.json está desactualizado: ejecuta python tools/export_openapi.py")
34                 return 1
35             print("docs/openapi.json está actualizado.")
36             return 0
37         DESTINO.write_text(actual, encoding="utf-8", newline="\n")
38         print(f"Escrito {DESTINO.relative_to(RAIZ)}")
39         return 0
40     
41     
42     if __name__ == "__main__":
43         sys.exit(main())
