22     def crear_parser() -> argparse.ArgumentParser:
23         p = argparse.ArgumentParser(
24             prog="python -m src.adapters.inbound.simulator",
25             description="Sensor simulado de calidad de agua para TruchaSmart",
26         )
27         p.add_argument("--estanque", type=int, required=True, help="id del estanque")
28         p.add_argument("--escenario", choices=ESCENARIOS, default="normal")
29         p.add_argument("--intervalo", type=float, default=5.0, help="segundos entre lecturas")
30         p.add_argument("--cantidad", type=int, default=0, help="n.º de lecturas (0 = sin fin)")
31         p.add_argument("--url", default=os.getenv("SIM_URL", "http://localhost:8000"))
32         p.add_argument("--usuario", default=os.getenv("SIM_USUARIO", "operario"))
33         p.add_argument("--password", default=os.getenv("SIM_PASSWORD", "Oper123!"))
34         p.add_argument("--semilla", type=int, default=None, help="semilla aleatoria reproducible")
35         return p

38     def ejecutar(args: argparse.Namespace, cliente: httpx.Client, dormir=time.sleep) -> int:
39         """Autentica y envía lecturas. Devuelve el n.º de lecturas aceptadas."""
40         r = cliente.post(
41             "/api/auth/login", json={"nombre_usuario": args.usuario, "password": args.password}
42         )
43         if r.status_code != 200:
44             print(f"Error de autenticación ({r.status_code}): {r.text}", file=sys.stderr)
45             return 0
46         cabeceras = {"Authorization": f"Bearer {r.json()['access_token']}"}
47         aceptadas = 0
48         for n, m in enumerate(generar(args.escenario, args.semilla), start=1):
49             cuerpo = {
50                 "estanque_id": args.estanque,
51                 "temperatura_c": m.temperatura_c,
52                 "oxigeno_mg_l": m.oxigeno_mg_l,
53                 "ph": m.ph,
54                 "fuente": "SENSOR",
55             }
56             try:
57                 resp = cliente.post("/api/lecturas", json=cuerpo, headers=cabeceras)
58             except httpx.HTTPError as e:
59                 print(f"[{n}] sin conexión con {args.url}: {e}", file=sys.stderr)
60             else:
61                 if resp.status_code == 201:
62                     aceptadas += 1
63                     datos = resp.json()
64                     alertas = datos.get("alertas", [])
65                     extra = f" · {len(alertas)} alerta(s)" if alertas else ""
66                     print(
67                         f"[{n}] estanque {args.estanque}: T={m.temperatura_c} °C, "
68                         f"OD={m.oxigeno_mg_l} mg/L, pH={m.ph}{extra}"
69                     )
70                 else:
71                     print(f"[{n}] rechazada ({resp.status_code}): {resp.text}", file=sys.stderr)
72             if args.cantidad and n >= args.cantidad:
73                 break
74             dormir(args.intervalo)
75         return aceptadas

78     def main(argv: list[str] | None = None) -> int:
79         args = crear_parser().parse_args(argv)
80         if hasattr(sys.stdout, "reconfigure"):
81             sys.stdout.reconfigure(encoding="utf-8")
82         print(f"Simulador → {args.url} · estanque {args.estanque} · escenario {args.escenario}")
83         try:
84             with httpx.Client(base_url=args.url, timeout=10) as cliente:
85                 ejecutar(args, cliente)
86         except KeyboardInterrupt:
87             print("\nSimulador detenido.")
88         return 0
