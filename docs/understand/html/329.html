35                 f.estanque,
36                 f.medido_en.strftime("%Y-%m-%d %H:%M:%S"),

66     class GeneradorReportesArchivo:
67         def csv(self, reporte: Reporte) -> bytes:
68             salida = io.StringIO(newline="")
69             escritor = csv.writer(salida, lineterminator="\r\n")
70             if reporte.tipo is TipoReporte.CALIDAD_AGUA:
71                 escritor.writerow(COLUMNAS_CALIDAD)
72                 escritor.writerows(_filas_calidad(reporte))
73             else:
74                 escritor.writerow(COLUMNAS_PRODUCCION)
75                 escritor.writerows(_filas_produccion(reporte))
76             # utf-8-sig: Excel abre correctamente las tildes.
77             return salida.getvalue().encode("utf-8-sig")
78     
79         def pdf(self, reporte: Reporte) -> bytes:
80             buffer = io.BytesIO()
81             doc = SimpleDocTemplate(
82                 buffer,
83                 pagesize=landscape(A4),
84                 leftMargin=1.5 * cm,
85                 rightMargin=1.5 * cm,
86                 topMargin=1.5 * cm,
87                 bottomMargin=1.5 * cm,
88                 title=f"TruchaSmart - {reporte.titulo}",
89                 author="TruchaSmart",
90             )
91             estilos = getSampleStyleSheet()
92             partes = [
93                 Paragraph(f"TruchaSmart · {reporte.titulo}", estilos["Title"]),
94                 Paragraph(
95                     f"Periodo: {reporte.desde:%d/%m/%Y} al {reporte.hasta:%d/%m/%Y} (UTC) · "
96                     f"Generado el {reporte.generado_en:%d/%m/%Y %H:%M} UTC por {reporte.generado_por}",
97                     estilos["Normal"],
98                 ),
99                 Spacer(1, 0.4 * cm),
100             ]
101             if reporte.tipo is TipoReporte.CALIDAD_AGUA:
102                 partes += self._pdf_calidad(reporte, estilos)
103             else:
104                 partes += self._pdf_produccion(reporte, estilos)
105             partes += [
106                 Spacer(1, 0.4 * cm),
107                 Paragraph(
108                     "Umbrales y tablas: valores de referencia parametrizables (config/*.yaml), "
109                     "a validar con el técnico de la piscigranja.",
110                     estilos["Italic"],
111                 ),
112             ]
113             doc.build(partes)
114             return buffer.getvalue()

118             tabla = Table(datos, colWidths=anchos, repeatRows=1)
119             tabla.setStyle(
120                 TableStyle(
121                     [
122                         ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#0b6e8a")),
123                         ("TEXTCOLOR", (0, 0), (-1, 0), colors.white),
124                         ("FONTSIZE", (0, 0), (-1, -1), 8),
125                         ("GRID", (0, 0), (-1, -1), 0.25, colors.HexColor("#d5dee5")),
126                         (
127                             "ROWBACKGROUNDS",
128                             (0, 1),
129                             (-1, -1),
130                             [colors.white, colors.HexColor("#f4f7f9")],
131                         ),
132                     ]

137         def _pdf_calidad(self, reporte: Reporte, estilos):
138             def est(t):
139                 return "—" if t is None else f"{t[0]:.2f} / {t[1]:.2f} / {t[2]:.2f}"
140     
141             resumen = [["Estanque", "Lecturas", "T °C (mín/prom/máx)", "OD mg/L (mín/prom/máx)",
142                         "pH (mín/prom/máx)", "Alertas P", "Alertas C", "Atendidas"]]  # fmt: skip
143             for r in reporte.resumen_calidad:
144                 resumen.append(
145                     [
146                         r.estanque,
147                         r.lecturas,
148                         est(r.temperatura),
149                         est(r.oxigeno),
150                         est(r.ph),
151                         r.alertas_precaucion,
152                         r.alertas_criticas,
153                         r.alertas_atendidas,
154                     ]
155                 )
156             partes = [Paragraph("Resumen por estanque", estilos["Heading2"]), self._tabla(resumen)]
157             detalle = [["Estanque", "Fecha UTC", "T °C", "OD mg/L", "pH", "Fuente"]]
158             for f in reporte.filas_calidad[:MAX_FILAS_PDF]:
159                 detalle.append([
160                     f.estanque,
161                     f.medido_en.strftime("%d/%m/%Y %H:%M"),
162                     f"{f.temperatura_c:.2f}{NIVEL_CORTO[f.niveles[Parametro.TEMPERATURA].value]}",
163                     f"{f.oxigeno_mg_l:.2f}{NIVEL_CORTO[f.niveles[Parametro.OXIGENO_DISUELTO].value]}",
164                     f"{f.ph:.2f}{NIVEL_CORTO[f.niveles[Parametro.PH].value]}",
165                     f.fuente.title(),
166                 ])  # fmt: skip
167             partes += [
168                 Spacer(1, 0.4 * cm),
169                 Paragraph(
170                     f"Detalle de lecturas ({len(reporte.filas_calidad)}) · (P) precaución, (C) crítica",
171                     estilos["Heading2"],
172                 ),
173             ]
174             if len(reporte.filas_calidad) > MAX_FILAS_PDF:
175                 partes.append(
176                     Paragraph(
177                         f"Se muestran las primeras {MAX_FILAS_PDF} lecturas; "
178                         "el detalle completo está en el CSV.",
179                         estilos["Normal"],
180                     )
181                 )
182             partes.append(
183                 self._tabla(detalle)
184                 if len(detalle) > 1
185                 else Paragraph("Sin lecturas.", estilos["Normal"])
186             )
187             return partes

189         def _pdf_produccion(self, reporte: Reporte, estilos):

191                       "Mort. total", "Vivos", "Peso g", "Biomasa kg", "Alim. periodo kg",
192                       "Alim. total kg", "FCR"]]  # fmt: skip
193             for fila in _filas_produccion(reporte):
194                 datos.append(fila)
195             if len(datos) == 1:
196                 return [Paragraph("Sin lotes vigentes en el periodo.", estilos["Normal"])]
197             return [
198                 Paragraph("Producción por lote", estilos["Heading2"]),
199                 self._tabla(datos),
200                 Paragraph(
201                     "Biomasa = peces vivos × peso medio actual · FCR = alimento total / ganancia de "
202                     'biomasa ("no calculable" si la ganancia es 0).',
203                     estilos["Normal"],
204                 ),
205             ]
