1     """Biomasa y factor de conversión alimenticia (FCR) de un lote (RF-06, brecha B-08).
2     
3       peces vivos      = peces sembrados - mortalidad acumulada
4       biomasa (kg)     = peces vivos x peso medio actual (g) / 1000
5       ganancia (kg)    = biomasa actual - biomasa inicial (peces sembrados x peso medio inicial)
6       FCR              = alimento total suministrado (kg) / ganancia de biomasa (kg)
7     Si la ganancia es 0 (o negativa) el FCR es "no calculable": no se divide ni se lanza error.
8     """
9     
10     from __future__ import annotations
11     
12     from dataclasses import dataclass
13     
14     from src.domain.errores import ErrorValidacion
15     from src.domain.model.lote import Lote
16     
17     NO_CALCULABLE = "no calculable"
18     
19     
20     @dataclass(frozen=True)
21     class ResumenProduccion:
22         lote_id: int
23         codigo: str
24         peces_sembrados: int
25         mortalidad_total: int
26         peces_vivos: int
27         supervivencia_pct: float
28         peso_medio_inicial_g: float
29         peso_medio_actual_g: float
30         biomasa_inicial_kg: float
31         biomasa_kg: float
32         ganancia_biomasa_kg: float
33         alimento_total_kg: float
34         fcr: float | None
35         fcr_detalle: str
36     
37         @property
38         def fcr_texto(self) -> str:
39             return NO_CALCULABLE if self.fcr is None else f"{self.fcr:.2f}"
40     
41     
42     def peces_vivos(lote: Lote, mortalidad_total: int) -> int:
43         if mortalidad_total < 0:
44             raise ErrorValidacion("cantidad", "La mortalidad acumulada no puede ser negativa")
45         if mortalidad_total > lote.num_peces_inicial:
46             raise ErrorValidacion(
47                 "cantidad",
48                 f"La mortalidad acumulada ({mortalidad_total}) supera los peces sembrados "
49                 f"({lote.num_peces_inicial})",
50             )
51         return lote.num_peces_inicial - mortalidad_total
52     
53     
54     def biomasa_kg(peces: int, peso_medio_g: float) -> float:
55         return round(peces * peso_medio_g / 1000, 3)
56     
57     
58     def calcular_fcr(alimento_total_kg: float, ganancia_kg: float) -> tuple[float | None, str]:
59         """Devuelve (FCR, explicación). FCR es None cuando no es calculable."""
60         if alimento_total_kg < 0:
61             raise ErrorValidacion("cantidad_kg", "El alimento total no puede ser negativo")
62         if ganancia_kg == 0:
63             return None, "No calculable: la ganancia de biomasa es 0 kg"
64         if ganancia_kg < 0:
65             return None, f"No calculable: la biomasa disminuyó {abs(ganancia_kg):.3f} kg"
66         fcr = round(alimento_total_kg / ganancia_kg, 2)
67         return fcr, (
68             f"FCR = {alimento_total_kg:.3f} kg de alimento / {ganancia_kg:.3f} kg de ganancia "
69             f"= {fcr:.2f}"
70         )
71     
72     
73     def resumen_produccion(
74         lote: Lote, mortalidad_total: int, alimento_total_kg: float
75     ) -> ResumenProduccion:
76         vivos = peces_vivos(lote, mortalidad_total)
77         inicial = biomasa_kg(lote.num_peces_inicial, lote.peso_medio_inicial_g)
78         actual = biomasa_kg(vivos, lote.peso_medio_actual_g)
79         ganancia = round(actual - inicial, 3)
80         alimento = round(alimento_total_kg, 3)
81         fcr, detalle = calcular_fcr(alimento, ganancia)
82         return ResumenProduccion(
83             lote_id=lote.id,
84             codigo=lote.codigo,
85             peces_sembrados=lote.num_peces_inicial,
86             mortalidad_total=mortalidad_total,
87             peces_vivos=vivos,
88             supervivencia_pct=round(vivos / lote.num_peces_inicial * 100, 2),
89             peso_medio_inicial_g=lote.peso_medio_inicial_g,
90             peso_medio_actual_g=lote.peso_medio_actual_g,
91             biomasa_inicial_kg=inicial,
92             biomasa_kg=actual,
93             ganancia_biomasa_kg=ganancia,
94             alimento_total_kg=alimento,
95             fcr=fcr,
96             fcr_detalle=detalle,
97         )
