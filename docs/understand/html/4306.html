39     class RepositorioEstanqueSQL:
40         def __init__(self, sesion: Session) -> None:
41             self._s = sesion
42     
43         def guardar(self, e: Estanque) -> Estanque:
44             fila = self._s.get(EstanqueORM, e.id) if e.id else None
45             if fila is None:
46                 fila = EstanqueORM()
47                 self._s.add(fila)
48             fila.nombre, fila.tipo = e.nombre, e.tipo.value
49             fila.volumen_m3, fila.ubicacion = e.volumen_m3, e.ubicacion
50             fila.activo, fila.creado_en = e.activo, e.creado_en
51             self._s.flush()
52             return _estanque(fila)
53     
54         def por_id(self, estanque_id: int) -> Estanque | None:
55             fila = self._s.get(EstanqueORM, estanque_id)
56             return _estanque(fila) if fila else None
57     
58         def por_nombre(self, nombre: str) -> Estanque | None:
59             fila = self._s.scalar(select(EstanqueORM).where(EstanqueORM.nombre == nombre))
60             return _estanque(fila) if fila else None
61     
62         def listar(self, solo_activos: bool = True) -> list[Estanque]:
63             consulta = select(EstanqueORM).order_by(EstanqueORM.nombre)
64             if solo_activos:
65                 consulta = consulta.where(EstanqueORM.activo.is_(True))
66             return [_estanque(f) for f in self._s.scalars(consulta)]
