17     class ModeloLogistico:
18         def __init__(self, datos: dict) -> None:
19             self._version = datos["version"]
20             self._caracteristicas = list(datos["caracteristicas"])
21             self._media = [float(v) for v in datos["media"]]
22             self._escala = [float(v) or 1.0 for v in datos["escala"]]
23             self._coef = [float(v) for v in datos["coeficientes"]]
24             self._intercepto = float(datos["intercepto"])
25             if not (
26                 len(self._caracteristicas) == len(self._media) == len(self._escala) == len(self._coef)
27             ):
28                 raise ValueError("El archivo del modelo tiene vectores de distinto tamaño")
29     
30         @classmethod
31         def desde_archivo(cls, ruta: Path) -> ModeloLogistico:
32             return cls(json.loads(Path(ruta).read_text(encoding="utf-8")))
33     
34         @property
35         def version(self) -> str:
36             return self._version
37     
38         def predecir(self, x: dict[str, float]) -> ResultadoModelo:
39             aportes = {
40                 nombre: coef * (x[nombre] - media) / escala
41                 for nombre, media, escala, coef in zip(
42                     self._caracteristicas, self._media, self._escala, self._coef, strict=True
43                 )
44             }
45             z = self._intercepto + sum(aportes.values())
46             probabilidad = 1 / (1 + math.exp(-z)) if z > -700 else 0.0
47             return ResultadoModelo(probabilidad, aportes)
