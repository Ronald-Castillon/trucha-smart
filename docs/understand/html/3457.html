47         def registrar(
48             self, comando: ComandoRegistrarLectura, solicitante: Usuario
49         ) -> ResultadoRegistroLectura:
50             solicitante.exigir(Permiso.REGISTRAR_LECTURA)
51             self._estanque_activo(comando.estanque_id)
52             ahora = self._reloj.ahora()
53             medido_en = comando.medido_en or ahora
54             if medido_en.tzinfo is None:
55                 raise ErrorValidacion("medido_en", "La fecha de medición debe incluir zona horaria")
56             if medido_en > ahora + TOLERANCIA_FUTURO:
57                 raise ErrorValidacion("medido_en", "La fecha de medición no puede ser futura")
58             lectura = self._lecturas.guardar(
59                 LecturaAgua(
60                     estanque_id=comando.estanque_id,
61                     temperatura_c=comando.temperatura_c,
62                     oxigeno_mg_l=comando.oxigeno_mg_l,
63                     ph=comando.ph,
64                     fuente=comando.fuente,
65                     medido_en=medido_en,
66                     registrado_por=solicitante.id,
67                 )
68             )
69             evaluaciones = self._evaluador.evaluar(lectura)
70             nuevas = []
71             for ev in evaluaciones:
72                 if not ev.requiere_alerta:
73                     continue
74                 # No duplicar: una sola alerta abierta por estanque, parámetro y nivel.
75                 if self._alertas.abierta(lectura.estanque_id, ev.parametro, ev.nivel) is not None:
76                     continue
77                 alerta = self._alertas.guardar(
78                     Alerta(
79                         estanque_id=lectura.estanque_id,
80                         lectura_id=lectura.id,
81                         parametro=ev.parametro,
82                         nivel=ev.nivel,
83                         valor=ev.valor,
84                         mensaje=ev.mensaje,
85                         creada_en=ahora,
86                     )
87                 )
88                 nuevas.append(alerta)
89             for alerta in nuevas:
90                 self._notificador.notificar(alerta)
91             return ResultadoRegistroLectura(lectura, evaluaciones, nuevas)
