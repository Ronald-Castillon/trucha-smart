18     def cargar_dotenv(ruta: Path = RAIZ / ".env") -> None:
19         """Carga KEY=VALUE de .env sin sobrescribir variables ya definidas."""
20         if not ruta.is_file():
21             return
22         for linea in ruta.read_text(encoding="utf-8").splitlines():
23             linea = linea.strip()
24             if not linea or linea.startswith("#") or "=" not in linea:
25                 continue
26             clave, valor = linea.split("=", 1)
27             os.environ.setdefault(clave.strip(), valor.strip().strip('"').strip("'"))

30     def normalizar_url_bd(url: str) -> str:
31         """Render y otros proveedores entregan `postgres://` o `postgresql://`; SQLAlchemy necesita
32         el controlador explícito (psycopg 3)."""
33         for prefijo in ("postgres://", "postgresql://"):
34             if url.startswith(prefijo):
35                 return "postgresql+psycopg://" + url.removeprefix(prefijo)
36         return url

48     class ConfiguracionInsegura(RuntimeError):
49         """La configuración no es apta para producción (RNF-+, OWASP A02)."""

52     @dataclass(frozen=True)
53     class Configuracion:
54         database_url: str = "sqlite:///./truchasmart.db"
55         jwt_secret: str = JWT_SECRET_POR_DEFECTO
56         jwt_expira_minutos: int = 480
57         umbrales_path: Path = RAIZ / "config" / "umbrales.yaml"
58         alimentacion_path: Path = RAIZ / "config" / "alimentacion.yaml"
59         modelo_riesgo_path: Path = RAIZ / "models" / "riesgo" / "modelo_v1.json"  # RF-12, ADR-007
60         entorno: str = "desarrollo"  # APP_ENV: desarrollo | produccion
61         salud_timeout_s: float = (
62             1.0  # SALUD_TIMEOUT_S: máximo de la verificación de /api/salud (RFC #63)
63         )
64     
65         @property
66         def es_produccion(self) -> bool:
67             return self.entorno.lower() in ("produccion", "producción", "production")
68     
69         def validar(self) -> None:
70             """En producción no se arranca con la clave JWT de ejemplo ni con una clave corta."""
71             if self.es_produccion and (
72                 self.jwt_secret == JWT_SECRET_POR_DEFECTO
73                 or len(self.jwt_secret) < LONGITUD_MINIMA_SECRETO
74             ):
75                 raise ConfiguracionInsegura(
76                     f"APP_ENV=produccion exige JWT_SECRET propio de al menos "
77                     f'{LONGITUD_MINIMA_SECRETO} caracteres (p. ej. python -c "import secrets; '
78                     f'print(secrets.token_urlsafe(48))")'
79                 )
80     
81         @classmethod
82         def desde_entorno(cls) -> Configuracion:
83             cargar_dotenv()
84             e = os.environ
85             return cls(
86                 database_url=normalizar_url_bd(e.get("DATABASE_URL", cls.database_url)),
87                 jwt_secret=e.get("JWT_SECRET", cls.jwt_secret),
88                 jwt_expira_minutos=int(e.get("JWT_EXPIRA_MINUTOS", cls.jwt_expira_minutos)),
89                 umbrales_path=_ruta(e.get("UMBRALES_PATH", "config/umbrales.yaml")),
90                 alimentacion_path=_ruta(e.get("ALIMENTACION_PATH", "config/alimentacion.yaml")),
91                 modelo_riesgo_path=_ruta(e.get("MODELO_RIESGO_PATH", "models/riesgo/modelo_v1.json")),
92                 entorno=e.get("APP_ENV", cls.entorno),
93                 salud_timeout_s=float(e.get("SALUD_TIMEOUT_S", cls.salud_timeout_s)),
94             )

97     def cargar_umbrales(ruta: Path) -> ConjuntoUmbrales:
98         """Lee config/umbrales.yaml y construye los umbrales del dominio (ADR-005)."""
99         datos = yaml.safe_load(Path(ruta).read_text(encoding="utf-8"))
100     
101         def umbral(clave: str, parametro: Parametro) -> Umbral:
102             d = datos[clave]
103             normal = d["normal"]
104             precaucion = d["precaucion"]
105             return Umbral(
106                 parametro=parametro,
107                 unidad=d.get("unidad", ""),
108                 normal_min=normal[0],
109                 normal_max=normal[1],
110                 precaucion_min=precaucion[0],
111                 precaucion_max=precaucion[1],
112             )
113     
114         return ConjuntoUmbrales(
115             temperatura=umbral("temperatura", Parametro.TEMPERATURA),
116             oxigeno_disuelto=umbral("oxigeno_disuelto", Parametro.OXIGENO_DISUELTO),
117             ph=umbral("ph", Parametro.PH),
118         )

121     def cargar_alimentacion(ruta: Path) -> TablaAlimentacion:
122         """Lee config/alimentacion.yaml (tabla de tasas y factores, ADR-005)."""
123         datos = yaml.safe_load(Path(ruta).read_text(encoding="utf-8"))
124         niveles = {"normal": Nivel.NORMAL, "precaucion": Nivel.PRECAUCION, "critica": Nivel.CRITICA}
125     
126         def factores(clave: str) -> dict[Nivel, float]:
127             return {niveles[k]: float(v) for k, v in datos[clave].items()}
128     
129         return TablaAlimentacion(
130             tramos=tuple(
131                 TramoTasa(float(t["desde_g"]), t["hasta_g"], float(t["tasa_pct"]))
132                 for t in datos["tasas"]
133             ),
134             factor_temperatura=factores("factor_temperatura"),
135             factor_oxigeno=factores("factor_oxigeno"),
136         )
