1     """RNF-U: verifica el contraste WCAG 2.1 (1.4.3, AA >= 4.5:1) de cada par texto/fondo de la paleta.
2     
3     Lee las variables de :root en estilos.css y calcula la razón de contraste con la luminancia relativa
4     de WCAG: L = 0.2126 R + 0.7152 G + 0.0722 B (canales linealizados);
5     razón = (L1 + 0.05) / (L2 + 0.05).
6     Sale con código 1 si algún par usado para texto no alcanza 4.5:1.
7     
8     Uso: python tools/contraste_paleta.py [--markdown]
9     """
10     
11     from __future__ import annotations
12     
13     import re
14     import sys
15     from pathlib import Path
16     
17     CSS = Path(__file__).resolve().parents[1] / "src/adapters/inbound/web/static/estilos.css"
18     MINIMO_AA = 4.5
19     
20     # (texto, fondo, dónde se usa). "#fff" = blanco literal en estilos.css.
21     PARES = [
22         ("texto", "fondo", "texto general sobre el fondo de la página"),
23         ("texto", "superficie", "texto en tarjetas"),
24         ("suave", "superficie", "texto secundario (.suave, etiquetas de formulario)"),
25         ("suave", "fondo", "texto secundario fuera de tarjetas"),
26         ("#ffffff", "primario", "menú superior y botones"),
27         ("primario", "superficie", "botón secundario y enlaces"),
28         ("#ffffff", "verde", "etiqueta / semáforo NORMAL"),
29         ("texto", "amarillo", "etiqueta / semáforo PRECAUCIÓN (texto oscuro)"),
30         ("#ffffff", "rojo", "etiqueta / semáforo CRÍTICA, botón detener simulador"),
31         ("amarillo-texto", "superficie", "valor en nivel de precaución (.nivel-PRECAUCION)"),
32         ("rojo", "superficie", "valor crítico, predicción de riesgo, bitácora"),
33         ("verde", "superficie", "indicador 'en vivo'"),
34         ("rojo", "#fdecea", "mensaje de error"),
35         ("verde", "#e8f5ee", "mensaje de confirmación"),
36     ]
37     
38     
39     def variables(css: str) -> dict[str, str]:
40         raiz = re.search(r":root\s*\{(.*?)\}", css, re.S).group(1)
41         return dict(re.findall(r"--([\w-]+):\s*(#[0-9a-fA-F]{6})", raiz))
42     
43     
44     def luminancia(hexa: str) -> float:
45         canales = [int(hexa[i : i + 2], 16) / 255 for i in (1, 3, 5)]
46         lin = [c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4 for c in canales]
47         return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]
48     
49     
50     def contraste(a: str, b: str) -> float:
51         claro, oscuro = sorted((luminancia(a), luminancia(b)), reverse=True)
52         return (claro + 0.05) / (oscuro + 0.05)
53     
54     
55     def main() -> int:
56         sys.stdout.reconfigure(encoding="utf-8")
57         paleta = variables(CSS.read_text(encoding="utf-8"))
58     
59         def color(nombre: str) -> str:
60             return nombre if nombre.startswith("#") else paleta[nombre]
61     
62         filas, fallos = [], 0
63         for texto, fondo, uso in PARES:
64             r = contraste(color(texto), color(fondo))
65             ok = r >= MINIMO_AA
66             fallos += not ok
67             filas.append((texto, color(texto), fondo, color(fondo), r, ok, uso))
68         if "--markdown" in sys.argv:
69             print("| Texto | Fondo | Contraste | AA (≥ 4.5:1) | Uso |")
70             print("|---|---|---:|:---:|---|")
71             for t, tc, f, fc, r, ok, uso in filas:
72                 print(f"| {t} `{tc}` | {f} `{fc}` | {r:.2f}:1 | {'Sí' if ok else 'No'} | {uso} |")
73         else:
74             for t, tc, f, fc, r, ok, uso in filas:
75                 estado = "OK " if ok else "NO "
76                 print(f"{estado} {r:5.2f}:1  {t:<15} {tc}  sobre {f:<11} {fc}  ({uso})")
77             print(f"\n{len(filas) - fallos}/{len(filas)} pares cumplen WCAG 2.1 AA (>= {MINIMO_AA}:1)")
78         return 1 if fallos else 0
79     
80     
81     if __name__ == "__main__":
82         sys.exit(main())
