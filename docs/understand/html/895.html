11     def _validar_positivo(campo: str, valor: float, descripcion: str) -> None:
12         if valor is None or valor <= 0:
13             raise ErrorValidacion(campo, f"{descripcion} debe ser mayor que 0")

20         num_peces_inicial: int

28         @classmethod
29         def sembrar(
30             cls,
31             estanque_id: int,
32             codigo: str,
33             num_peces: int,
34             peso_medio_g: float,
35             fecha_siembra: date,
36             hoy: date,
37         ) -> Lote:
38             codigo = codigo.strip()

42             _validar_positivo("peso_medio_g", peso_medio_g, "El peso medio")
43             if fecha_siembra > hoy:
44                 raise ErrorValidacion("fecha_siembra", "La fecha de siembra no puede ser futura")
45             return cls(
46                 estanque_id=estanque_id,
47                 codigo=codigo,
48                 num_peces_inicial=num_peces,
49                 peso_medio_inicial_g=peso_medio_g,

51                 fecha_siembra=fecha_siembra,

54         def actualizar_peso(self, peso_medio_g: float) -> None:
55             """Registra el peso medio de la última biometría (muestreo)."""
56             self._exigir_activo()
57             _validar_positivo("peso_medio_g", peso_medio_g, "El peso medio")
58             self.peso_medio_actual_g = peso_medio_g
59     
60         def cerrar(self, fecha: date) -> None:
61             """Cierra el lote (cosecha o traslado); el estanque queda libre para otro lote."""
62             self._exigir_activo()
63             if fecha < self.fecha_siembra:
64                 raise ErrorValidacion("fecha_cierre", "La fecha de cierre es anterior a la siembra")
65             self.activo = False
66             self.fecha_cierre = fecha
67     
68         def _exigir_activo(self) -> None:
69             if not self.activo:
70                 raise Conflicto(f"El lote {self.codigo} está cerrado")
