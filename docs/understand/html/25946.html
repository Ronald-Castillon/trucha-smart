129     def ejecutar(comando: str, cwd: Path, env: dict[str, str]) -> tuple[int, str]:
130         r = subprocess.run(
131             ["bash", "-o", "pipefail", "-c", comando], cwd=cwd, env=env, capture_output=True,
132             text=True, encoding="utf-8", errors="replace", timeout=1200,
133         )  # fmt: skip
134         salida = ANSI.sub("", (r.stdout + r.stderr).rstrip())
135         return r.returncode, JWT.sub(lambda m: m.group(0)[:20] + "…(oculto)", salida)

138     def git(*args: str, cwd: Path) -> str:
139         return subprocess.run(["git", *args], cwd=cwd, capture_output=True, text=True).stdout.strip()

142     def main(argv: list[str] | None = None) -> int:
143         parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
144         parser.add_argument("--tag-dir", type=Path, required=True, help="checkout del tag v3.0-Inc3")

149         args = parser.parse_args(argv)

155         )
156         corrida, numero = os.getenv("GITHUB_RUN_ID", ""), os.getenv("GITHUB_RUN_NUMBER", "local")
157         enlace = f"{servidor}/{repo}/actions/runs/{corrida}" if corrida else "(ejecución local)"
158         dirs = {"tag": args.tag_dir, "tag1": args.tag_dir_1, "tag2": args.tag_dir_2}
159         versiones = {
160             clave: (
161                 d.resolve(),
162                 f"tag `{TAGS[clave]}` @ `{git('rev-parse', '--short', 'HEAD', cwd=d)}`",
163             )
164             for clave, d in dirs.items()
165             if d
166         }
167         versiones["main"] = (RAIZ, f"rama `main` @ `{git('rev-parse', '--short', 'HEAD', cwd=RAIZ)}`")
168         indice = []
169         for nombre, titulo, donde, comando in EVIDENCIAS:
170             if args.solo and nombre not in args.solo:
171                 continue
172             if donde not in versiones:
173                 continue
174             cwd, version = versiones[donde]
175             for clave, (ruta, _) in versiones.items():  # {tag1}, {tag2}, {tag}: carpeta de cada tag
176                 comando = comando.replace("{" + clave + "}", str(ruta))
177             env = {**os.environ, "PYTHONIOENCODING": "utf-8", "NO_COLOR": "1", "COLUMNS": "120"}
178             if donde.startswith("tag"):
179                 env["PYTHONPATH"] = str(cwd)
180             inicio = datetime.now(UTC)
181             codigo, salida = ejecutar(comando, cwd, env)
182             print(f"{'ok ' if codigo == 0 else 'ERR'} {nombre} ({codigo})", flush=True)
183             (args.salida / f"{nombre}.md").write_text(
184                 f"# {titulo}\n\n"
185                 f"| Ejecución | Valor |\n|---|---|\n"
186                 f"| Dónde | GitHub Actions · workflow «Evidencias del informe» · corrida #{numero} |\n"
187                 f"| Enlace | {enlace} |\n| Versión del código | {version} |\n"
188                 f"| Fecha (UTC) | {inicio:%d/%m/%Y %H:%M:%S} |\n| Código de salida | {codigo} |\n\n"
189                 f"```bash\n$ {comando}\n```\n\n```text\n{salida}\n```\n",
190                 encoding="utf-8",
191             )
192             indice.append(f"| [{nombre}]({nombre}.md) | {titulo} | {version} | {codigo} |")
193         (args.salida / "README.md").write_text(
194             "# Evidencias ejecutadas en GitHub Actions\n\n"
195             f"Corrida #{numero} del workflow «Evidencias del informe»: {enlace}\n\n"
196             "Cada página guarda el comando, su salida literal y el código de salida (0 = correcto). "

198             "y seguridad en ejecución, sobre main.\n\n"
199             "| Evidencia | Qué verifica | Versión | Código de salida |\n|---|---|---|---|\n"
200             + "\n".join(indice) + "\n",
201             encoding="utf-8",
202         )  # fmt: skip
