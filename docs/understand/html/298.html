26     @router.post("/lecturas", response_model=LecturaRegistradaSalida, status_code=201)
27     def registrar_lectura(
28         datos: LecturaEntrada,
29         usuario: Usuario = Depends(usuario_actual),
30         c: Contenedor = Depends(contenedor),
31     ) -> LecturaRegistradaSalida:
32         """Registra una lectura (formulario o sensor simulado). Valores físicamente imposibles
33         (pH fuera de 0–14, OD < 0, temperatura fuera de 0–40 °C) se rechazan con 422.
34     
35         La lectura se evalúa contra config/umbrales.yaml y, si un parámetro está en
36         PRECAUCIÓN o CRÍTICA, se crea una alerta (sin duplicar una abierta del mismo
37         estanque, parámetro y nivel)."""
38         resultado = c.lecturas.registrar(
39             ComandoRegistrarLectura(
40                 estanque_id=datos.estanque_id,
41                 temperatura_c=datos.temperatura_c,
42                 oxigeno_mg_l=datos.oxigeno_mg_l,
43                 ph=datos.ph,
44                 fuente=datos.fuente,
45                 medido_en=datos.medido_en,
46             ),
47             usuario,
48         )
49         c.confirmar()
50         return LecturaRegistradaSalida(
51             **LecturaSalida.de(resultado.lectura).model_dump(),
52             evaluacion=[EvaluacionSalida.de(e) for e in resultado.evaluaciones],
53             alertas=[AlertaSalida.de(a) for a in resultado.alertas_nuevas],
54         )

78     @router.get("/panel", response_model=list[EstadoEstanque], tags=["panel"])
79     def estado_panel(
80         request: Request,
81         puntos: int = Query(20, ge=0, le=200, description="Lecturas para la tendencia"),
82         usuario: Usuario = Depends(usuario_actual),
83         c: Contenedor = Depends(contenedor),
84     ) -> list[EstadoEstanque]:
85         """Estado de todos los estanques activos; el panel web lo consulta cada 1 s."""
86         sim = getattr(request.app.state, "simulador", None)
87         resultado = []
88         for e in c.estanques.listar(usuario):
89             estado = c.lecturas.estado(e.id, usuario)
90             recientes = c.lecturas.recientes(e.id, usuario, puntos) if puntos else []
91             simulacion = sim.de(e.id) if sim else None
92             pred = c.prediccion.predecir(e.id, usuario).principal
93             con_aviso = pred is not None and pred.estado.value in ("RIESGO", "YA_CRITICO")
94             resultado.append(
95                 EstadoEstanque(
96                     estanque_id=e.id,
97                     nombre=e.nombre,
98                     ultima_lectura=LecturaSalida.de(estado.lectura) if estado.lectura else None,
99                     niveles=estado.niveles,
100                     alertas_abiertas=[AlertaSalida.de(a) for a in estado.alertas_abiertas],
101                     # mismo cálculo que /riesgo, con el estado ya leído (RNF-P: menos consultas)
102                     riesgo=RiesgoSalida.de(riesgo_de(estado.niveles, estado.alertas_abiertas)),
103                     tendencia=[PuntoTendencia.de(x) for x in reversed(recientes)],
104                     simulacion=simulacion.escenario if simulacion else None,
105                     prediccion=pred.mensaje if con_aviso else None,
106                     prediccion_estado=pred.estado.value if pred else None,
107                 )
108             )
109         return resultado
