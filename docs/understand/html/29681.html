1     """Entrena el modelo de riesgo de evento crítico (RF-12, ADR-007) con datos sintéticos.
2     
3     1. Genera trayectorias de 36 h (una lectura cada 30 min) de temperatura, oxígeno disuelto y pH
4        con ciclo diario y ruido, y eventos aleatorios: hipoxia (el OD cae), calentamiento (la
5        temperatura sube) y falsas alarmas (una caída que se detiene antes del límite).
6     2. En cada trayectoria toma instantes t entre la hora 24 y la 30; las características se calculan
7        con la misma función del dominio (src.domain.service.riesgo_modelo.caracteristicas) sobre las
8        24 h previas, y la etiqueta es 1 si en (t, t + 6 h] el OD baja del límite o la temperatura lo
9        supera. Se descartan instantes ya críticos (el dominio los informa sin modelo).
10     3. Separa entrenamiento y prueba por trayectoria (sin fuga entre conjuntos), entrena una
11        regresión logística con características estandarizadas y la compara con la regla de
12        tendencia del Incremento 3 (ADR-006) sobre el mismo conjunto de prueba.
13     4. Exporta models/riesgo/modelo_v1.json (sin pickle) y models/riesgo/tarjeta_modelo_v1.md.
14     
15     Todo es reproducible con --semilla. Uso:
16       pip install -e ".[ml]"
17       python tools/entrenar_modelo_riesgo.py            # entrena y escribe el modelo
18       python tools/entrenar_modelo_riesgo.py --verificar # reentrena y compara con el archivo guardado
19     """

44     VERSION = "riesgo-v1"
45     DESTINO = RAIZ / "models" / "riesgo"
46     OD_LIMITE, T_LIMITE = 5.0, 18.0  # límites de precaución de config/umbrales.yaml
47     PASO = timedelta(minutes=30)
48     HORAS = 36
49     UMBRAL_OD = Umbral(Parametro.OXIGENO_DISUELTO, "mg/L", 7.0, None, OD_LIMITE, None)
50     UMBRAL_T = Umbral(Parametro.TEMPERATURA, "°C", 10.0, 16.0, 8.0, T_LIMITE)
51     T0 = datetime(2026, 1, 1, tzinfo=UTC)

54     def trayectoria(rnd: random.Random) -> list[tuple[datetime, float, float, float]]:
55         od0, t0, ph0 = rnd.uniform(6.8, 9.5), rnd.uniform(10.0, 14.5), rnd.uniform(6.8, 7.8)
56         fase = rnd.uniform(0, 24)
57         evento = rnd.choices(["ninguno", "hipoxia", "calor", "falsa_alarma"], [0.40, 0.30, 0.15, 0.15])[
58             0
59         ]
60         inicio = rnd.uniform(18, 33)
61         tasa = rnd.uniform(0.12, 0.8)
62         duracion_falsa = rnd.uniform(2, 5)
63         puntos = []
64         for i in range(int(HORAS * 2) + 1):
65             h = i * 0.5
66             ciclo = math.sin(2 * math.pi * (h + fase - 14) / 24)
67             od = od0 + 0.5 * ciclo + rnd.gauss(0, 0.12)
68             t = t0 + 0.8 * ciclo + rnd.gauss(0, 0.08)
69             ph = ph0 + rnd.gauss(0, 0.04)
70             d = max(0.0, h - inicio)
71             if evento == "hipoxia":
72                 od -= tasa * d
73             elif evento == "calor":
74                 t += tasa * 0.6 * d
75             elif evento == "falsa_alarma":
76                 od -= tasa * 0.6 * min(d, duracion_falsa)
77             puntos.append((T0 + i * PASO, round(t, 2), round(max(od, 0.0), 2), round(ph, 2)))
78         return puntos

81     def muestras(tray, id_tray):
82         filas = []
83         for k in range(48, 61, 2):  # t entre la hora 24 y la 30, cada hora
84             t = tray[k][0]
85             ventana = [p for p in tray if t - timedelta(hours=24) < p[0] <= t]
86             x = caracteristicas(ventana, OD_LIMITE, T_LIMITE)
87             if x is None or x["od_margen"] < 0 or x["t_margen"] < 0:
88                 continue
89             futuro = [p for p in tray if t < p[0] <= t + timedelta(hours=6)]
90             y = int(any(od < OD_LIMITE or temp > T_LIMITE for _, temp, od, _ in futuro))
91             regla = any(
92                 predecir(par, [(m, v) for m, v in serie], umb).estado
93                 in (EstadoPrediccion.RIESGO, EstadoPrediccion.YA_CRITICO)
94                 for par, serie, umb in (
95                     (Parametro.OXIGENO_DISUELTO, [(m, od) for m, _, od, _ in ventana], UMBRAL_OD),
96                     (Parametro.TEMPERATURA, [(m, temp) for m, temp, _, _ in ventana], UMBRAL_T),
97                 )
98             )
99             filas.append((id_tray, [x[c] for c in CARACTERISTICAS], y, int(regla)))
100         return filas

103     def metricas(y, pred, prob=None):
104         m = {
105             "precision": round(float(precision_score(y, pred, zero_division=0)), 4),
106             "recall": round(float(recall_score(y, pred, zero_division=0)), 4),
107             "f1": round(float(f1_score(y, pred, zero_division=0)), 4),
108         }
109         if prob is not None:
110             m["auc"] = round(float(roc_auc_score(y, prob)), 4)
111         return m

114     def entrenar(semilla: int, n: int) -> tuple[dict, dict]:
115         rnd = random.Random(semilla)
116         filas = [f for i in range(n) for f in muestras(trayectoria(rnd), i)]
117         ids = sorted({f[0] for f in filas})
118         random.Random(semilla + 1).shuffle(ids)
119         prueba_ids = set(ids[: len(ids) // 4])
120         entrena = [f for f in filas if f[0] not in prueba_ids]
121         prueba = [f for f in filas if f[0] in prueba_ids]
122         xe, ye = np.array([f[1] for f in entrena]), np.array([f[2] for f in entrena])
123         xp, yp = np.array([f[1] for f in prueba]), np.array([f[2] for f in prueba])
124         escalador = StandardScaler().fit(xe)
125         clf = LogisticRegression(C=1.0, max_iter=2000).fit(escalador.transform(xe), ye)
126         prob = clf.predict_proba(escalador.transform(xp))[:, 1]
127         modelo = {
128             "version": VERSION,
129             "algoritmo": "Regresión logística (scikit-learn) sobre características estandarizadas",
130             "caracteristicas": list(CARACTERISTICAS),
131             "media": [round(float(v), 6) for v in escalador.mean_],
132             "escala": [round(float(v), 6) for v in escalador.scale_],
133             "coeficientes": [round(float(v), 6) for v in clf.coef_[0]],
134             "intercepto": round(float(clf.intercept_[0]), 6),
135             "datos": "Sintéticos (no hay historial real etiquetado); ver tarjeta_modelo_v1.md",
136             "semilla": semilla,
137             "trayectorias": n,
138         }
139         # ejemplos para verificar que el adaptador reproduce exactamente al modelo entrenado
140         modelo["ejemplos_verificacion"] = [
141             {"x": dict(zip(CARACTERISTICAS, map(float, xp[i]), strict=True)),
142              "probabilidad": round(float(prob[i]), 6)}
143             for i in range(0, len(xp), max(1, len(xp) // 5))
144         ][:5]  # fmt: skip
145         informe = {
146             "muestras": {"entrenamiento": len(entrena), "prueba": len(prueba),
147                          "positivos_prueba": int(yp.sum()), "trayectorias_prueba": len(prueba_ids)},
148             "modelo_umbral_0_5": metricas(yp, (prob >= 0.5).astype(int), prob),
149             "modelo_umbral_alto_0_6": metricas(yp, (prob >= 0.6).astype(int)),
150             "regla_tendencia_inc3": metricas(yp, np.array([f[3] for f in prueba])),
151         }  # fmt: skip
152         return modelo, informe

155     def tarjeta(modelo: dict, informe: dict) -> str:
156         m, r, a = (
157             informe["modelo_umbral_0_5"],
158             informe["regla_tendencia_inc3"],
159             informe["modelo_umbral_alto_0_6"],
160         )
161         s = informe["muestras"]
162         coef = sorted(
163             zip(modelo["caracteristicas"], modelo["coeficientes"], strict=True),
164             key=lambda c: -abs(c[1]),
165         )
166         return "\n".join([
167             f"# Tarjeta del modelo `{modelo['version']}` (RF-12, ADR-007)",
168             "",
169             "Generada por `tools/entrenar_modelo_riesgo.py`; no se edita a mano.",
170             "",
171             "## Uso previsto",
172             "Estimar la probabilidad de que un estanque llegue a condición crítica (OD < 5 mg/L o",
173             "temperatura > 18 °C) en las próximas 6 h, para priorizar revisiones. Es apoyo a la",
174             "decisión: no reemplaza las alertas por umbral ni el criterio del técnico.",
175             "",
176             "## Datos",
177             f"- **Sintéticos**: {modelo['trayectorias']} trayectorias de 36 h "
178             f"(semilla {modelo['semilla']}) con",
179             "  ciclo diario, ruido y eventos de hipoxia, calentamiento y falsas alarmas.",
180             f"- Entrenamiento: {s['entrenamiento']} muestras; prueba: {s['prueba']} muestras de",
181             f"  {s['trayectorias_prueba']} trayectorias no vistas ({s['positivos_prueba']} positivas).",
182             "- **Limitación**: no hay historial real etiquetado de la piscigranja. Las métricas miden",
183             "  qué tan bien el modelo aprende el generador sintético, no su desempeño en campo.",
184             "",
185             "## Métricas en el conjunto de prueba",
186             "",
187             "| Método | Precisión | Recall | F1 | AUC |",
188             "|---|---:|---:|---:|---:|",
189             f"| Modelo, umbral 0.5 | {m['precision']} | {m['recall']} | {m['f1']} | {m['auc']} |",
190             f"| Modelo, umbral ALTO 0.6 | {a['precision']} | {a['recall']} | {a['f1']} | — |",
191             f"| Regla de tendencia del Inc. 3 (ADR-006) | {r['precision']} | {r['recall']} "
192             f"| {r['f1']} | — |",

199             f"| intercepto | {modelo['intercepto']:+.3f} |",
200             "",
201             "## Antes de usarlo en producción",
202             "Reentrenar con lecturas reales etiquetadas (eventos críticos registrados en las alertas),",
203             "validar con el técnico de la piscigranja y comparar otra vez con la regla del Inc. 3.",
204             "",
205         ])  # fmt: skip

208     def main(argv: list[str] | None = None) -> int:
209         ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawTextHelpFormatter)
210         ap.add_argument("--semilla", type=int, default=2026)
211         ap.add_argument("--trayectorias", type=int, default=1500)
212         ap.add_argument("--verificar", action="store_true", help="reentrena y compara con el archivo")
213         args = ap.parse_args(argv)
214         sys.stdout.reconfigure(encoding="utf-8")
215         modelo, informe = entrenar(args.semilla, args.trayectorias)
216         print(json.dumps(informe, ensure_ascii=False, indent=2))
217         ruta = DESTINO / "modelo_v1.json"
218         if args.verificar:
219             guardado = json.loads(ruta.read_text(encoding="utf-8"))
220             claves = ("media", "escala", "coeficientes", "intercepto")
221             # tolerancia: el optimizador puede diferir en el último decimal entre sistemas operativos
222             iguales = all(
223                 math.isclose(a, b, abs_tol=1e-4)
224                 for k in claves
225                 for a, b in zip(
226                     guardado[k] if isinstance(guardado[k], list) else [guardado[k]],
227                     modelo[k] if isinstance(modelo[k], list) else [modelo[k]],
228                     strict=True,
229                 )
230             )
231             mejor = informe["modelo_umbral_0_5"]["f1"] > informe["regla_tendencia_inc3"]["f1"]
232             print(
233                 f"Modelo reproducible: {'sí' if iguales else 'NO'} · F1 del modelo mayor que el de "
234                 f"la regla del Inc. 3: {'sí' if mejor else 'NO'}"
235             )
236             return 0 if iguales and mejor else 1
237         DESTINO.mkdir(parents=True, exist_ok=True)
238         modelo["metricas_prueba"] = informe
239         ruta.write_text(
240             json.dumps(modelo, ensure_ascii=False, indent=2) + "\n", encoding="utf-8", newline="\n"
241         )
242         (DESTINO / "tarjeta_modelo_v1.md").write_text(
243             tarjeta(modelo, informe), encoding="utf-8", newline="\n"
244         )
245         print(f"Escrito {ruta.relative_to(RAIZ)} y tarjeta_modelo_v1.md")
246         return 0
