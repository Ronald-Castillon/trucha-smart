20     class FechaHoraUTC(TypeDecorator):
21         """Guarda fechas en UTC y las devuelve siempre con zona horaria (SQLite no la conserva)."""
22     
23         impl = DateTime(timezone=True)
24         cache_ok = True

26         def process_bind_param(self, valor: datetime | None, dialect):
27             if valor is None:
28                 return None
29             if valor.tzinfo is None:
30                 valor = valor.replace(tzinfo=UTC)
31             valor = valor.astimezone(UTC)
32             return valor.replace(tzinfo=None) if dialect.name == "sqlite" else valor

34         def process_result_value(self, valor: datetime | None, dialect):
35             if valor is None:
36                 return None
37             return valor.replace(tzinfo=UTC) if valor.tzinfo is None else valor.astimezone(UTC)

40     def crear_motor(url: str) -> Engine:
41         if url.startswith("sqlite"):
42             opciones: dict = {"connect_args": {"check_same_thread": False, "timeout": 15}}
43             en_memoria = url in ("sqlite://", "sqlite:///:memory:")
44             if en_memoria:
45                 opciones["poolclass"] = StaticPool
46             else:
47                 # Defecto #39: el pool por defecto (5 + 10) se agotaba con los 40 hilos de cada worker.
48                 opciones.update(pool_size=20, max_overflow=30)
49             motor = create_engine(url, **opciones)
50     
51             @event.listens_for(motor, "connect")
52             def _configurar(conexion, _):
53                 conexion.execute("PRAGMA foreign_keys = ON")
54                 if not en_memoria:
55                     # Defecto #39: con el diario por defecto, una escritura bloquea a los lectores y
56                     # bajo carga aparecían 500 "database is locked". WAL permite leer mientras se
57                     # escribe; busy_timeout espera el turno en lugar de fallar.
58                     # busy_timeout primero: varios workers arrancan a la vez y cambian el modo.
59                     conexion.execute("PRAGMA busy_timeout = 15000")
60                     conexion.execute("PRAGMA journal_mode = WAL")
61     
62             return motor
63         # pool_pre_ping descarta conexiones rotas: tras una caída de PostgreSQL reconecta (RNF-R)
64         # 4 workers x (10 + 15) = 100 conexiones como máximo, el límite por defecto de PostgreSQL.
65         return create_engine(
66             url,
67             pool_pre_ping=True,
68             pool_size=10,
69             max_overflow=15,
70             connect_args={"connect_timeout": 3},
71         )

107         def _consultar(self) -> bool:
108             try:
109                 with self.motor.connect() as conexion:
110                     conexion.execute(text("SELECT 1"))
111                 return True
112             except SQLAlchemyError:
113                 return False

115         def disponible(self) -> bool:
116             """`SELECT 1` con tiempo máximo propio (usado por /api/salud, RNF-R).
117     
118             RFC #63 (defecto #51): si la base acepta la conexión pero no responde, la consulta podía
119             tardar más de 4 s y la sonda expiraba sin recibir el 503. Ahora se espera como máximo
120             `salud_timeout_s`; si la sonda anterior sigue colgada se responde «no disponible» sin
121             lanzar otra.
122             """
123             if self._sonda is not None and not self._sonda.done():
124                 return False
125             self._sonda = _SONDAS.submit(self._consultar)
126             try:
127                 return self._sonda.result(timeout=self.salud_timeout_s)
128             except TiempoAgotado:
129                 return False
